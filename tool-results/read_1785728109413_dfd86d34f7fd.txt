    60→function EvalStatCards({ evaluations, evalBatches, evalLoading }: {
    61→  evaluations: Evaluation[];
    62→  evalBatches: EvalBatch[];
    63→  evalLoading: boolean;
    64→}) {
    65→  const { locale } = useI18n();
    66→  const totalEvals = evaluations.length;
    67→  const totalBatches = evalBatches.length;
    68→
    69→  const avgCoverage = useMemo(() => {
    70→    const withCoverage = evaluations.filter(e => e.coverage != null);
    71→    if (withCoverage.length === 0) return 0;
    72→    return withCoverage.reduce((sum, e) => sum + (e.coverage ?? 0), 0) / withCoverage.length;
    73→  }, [evaluations]);
    74→
    75→  const highCoverageCount = evaluations.filter(e => (e.coverage ?? 0) >= 80).length;
    76→
    77→  const highCoveragePct = totalEvals > 0 ? (highCoverageCount / totalEvals) * 100 : 0;
    78→
    79→  const completionRate = useMemo(() => {
    80→    if (totalEvals === 0) return 0;
    81→    const completedCount = evaluations.filter(e => {
    82→      const hasPdb = (e.pdbStructures?.length ?? 0) > 0;
    83→      const hasBlast = (e.blastResults?.length ?? 0) > 0;
    84→      const hasReport = !!e.report;
    85→      return hasPdb && hasBlast && hasReport;
    86→    }).length;
    87→    return (completedCount / totalEvals) * 100;
    88→  }, [evaluations, totalEvals]);
    89→
    90→  return (
    91→    <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 sm:gap-3 p-2 sm:p-3 [grid-auto-rows:1fr] min-w-0 stagger-list">
    92→      {/* Eval Targets — completion rate ring */}
    93→      <StatCard
    94→        title={locale === 'zh' ? '评估靶点' : 'Eval Targets'}
    95→        value={totalEvals}
    96→        icon={<FlaskConical className="h-3.5 w-3.5 text-white" />}
    97→        color="bg-gradient-to-br from-[#2d8f8f] to-[#1a6b6b]"
    98→        glowColor="#2d8f8f"
    99→        subtitle={locale === 'zh' ? `${totalBatches} 个批次 · ${completionRate.toFixed(0)}% 完成` : `${totalBatches} batch${totalBatches !== 1 ? 'es' : ''} · ${completionRate.toFixed(0)}% done`}
   100→        loading={evalLoading}
   101→        delay={0}
   102→        borderColor="#2d8f8f"
   103→        tooltip={locale === 'zh' ? `评估靶点: ${totalEvals} (${completionRate.toFixed(0)}% 完成, ${totalBatches} 个批次)` : `Eval Targets: ${totalEvals} (${completionRate.toFixed(0)}% complete, ${totalBatches} batches)`}
   104→      >
   105→        <CircularProgress value={completionRate} max={100} color="#2d8f8f" size={28} />
   106→      </StatCard>
   107→
   108→      {/* Batches — mini bar showing batch density */}
   109→      <StatCard
   110→        title={locale === 'zh' ? '批量评估' : 'Batches'}
   111→        value={totalBatches}
   112→        icon={<Database className="h-3.5 w-3.5 text-white" />}
   113→        color="bg-gradient-to-br from-[#c9872e] to-[#a06b1a]"
   114→        glowColor="#c9872e"
   115→        subtitle={locale === 'zh' ? `${totalEvals} 个评估 · 平均 ${(totalEvals / Math.max(totalBatches, 1)).toFixed(1)}` : `${totalEvals} evals · ${(totalEvals / Math.max(totalBatches, 1)).toFixed(1)} avg`}
   116→        loading={evalLoading}
   117→        delay={80}
   118→        borderColor="#c9872e"
   119→        tooltip={locale === 'zh' ? `批次: ${totalBatches} (${totalEvals} 个评估, 平均 ${(totalEvals / Math.max(totalBatches, 1)).toFixed(1)} 评估/批次)` : `Batches: ${totalBatches} (${totalEvals} evaluations, avg ${(totalEvals / Math.max(totalBatches, 1)).toFixed(1)} evals/batch)`}
   120→      >
   121→        <MiniBar value={totalBatches > 0 ? Math.min((totalBatches / Math.max(totalEvals, 1)) * 100 * 3, 100) : 0} max={100} color="#c9872e" width={40} height={5} />
   122→      </StatCard>
   123→
   124→      {/* Avg Coverage — coverage ring */}
   125→      <StatCard
   126→        title={locale === 'zh' ? '平均覆盖率' : 'Avg Coverage'}
   127→        value={avgCoverage}
   128→        suffix="%"
   129→        decimals={0}
   130→        icon={<Target className="h-3.5 w-3.5 text-white" />}
   131→        color="bg-gradient-to-br from-[#7c5cbf] to-[#5a3d99]"
   132→        glowColor="#7c5cbf"
   133→        subtitle={locale === 'zh' ? `${highCoverageCount} 高 (≥80%)` : `${highCoverageCount} high (≥80%)`}
   134→        loading={evalLoading}
   135→        delay={160}
   136→        borderColor="#7c5cbf"
   137→        tooltip={locale === 'zh' ? `平均覆盖率: ${avgCoverage.toFixed(1)}% (${highCoverageCount} 高 ≥80%)` : `Avg Coverage: ${avgCoverage.toFixed(1)}% (${highCoverageCount} high ≥80%)`}
   138→      >
   139→        <CircularProgress value={avgCoverage} max={100} color="#7c5cbf" size={28} />
   140→      </StatCard>
   141→
   142→      {/* ≥80% Coverage — mini bar */}
   143→      <StatCard
   144→        title={locale === 'zh' ? '≥80% 覆盖率' : '≥80% Coverage'}
   145→        value={highCoverageCount}
   146→        icon={<CheckCircle2 className="h-3.5 w-3.5 text-white" />}
   147→        color="bg-gradient-to-br from-[#16a34a] to-[#0d7a35]"
   148→        glowColor="#16a34a"
   149→        subtitle={totalEvals > 0 ? (locale === 'zh' ? `${((highCoverageCount / totalEvals) * 100).toFixed(0)}% 占比` : `${((highCoverageCount / totalEvals) * 100).toFixed(0)}% of total`) : (locale === 'zh' ? '暂无数据' : 'No data')}
   150→        loading={evalLoading}
   151→        delay={240}
   152→        borderColor="#16a34a"
   153→        tooltip={locale === 'zh' ? `高覆盖: ${highCoverageCount} 个靶点 ≥80% (${totalEvals > 0 ? ((highCoverageCount / totalEvals) * 100).toFixed(0) : 0}%)` : `High Coverage: ${highCoverageCount} targets ≥80% (${totalEvals > 0 ? ((highCoverageCount / totalEvals) * 100).toFixed(0) : 0}%)`}
   154→      >
   155→        <MiniBar value={highCoveragePct} max={100} color="#16a34a" width={40} height={5} />
   156→      </StatCard>
   157→    </div>
   158→  );
   159→}

   161→// ─── BatchDetailView ─────────────────────────────────────────────────────────

   168→/** Parse `commonPdbIds` (stored as a JSON-stringified array) into a string[]. */
   169→function parseCommonPdbIds(raw: string | null | undefined): string[] {
   170→  if (!raw) return [];
   171→  try {
   172→    const parsed = JSON.parse(raw);
   173→    if (Array.isArray(parsed)) return parsed.filter(Boolean) as string[];
   174→  } catch {
   175→    return raw.split(/[\s,]+/).filter(Boolean);
   176→  }
   177→  return [];
   178→}

   190→// ─── BatchCommonPdbView ────────────────────────────────────────────────────
   191→// Simplified batch view: shows the common (shared) PDB list in the middle,
   192→// mirroring the single-eval PDB table layout. The batch's combined report
   193→// is shown in the right detail panel (same as single eval).
   194→function BatchCommonPdbView({
   195→  batchId,
   196→  allEvals,
   197→  evalBatches,
   198→  evalBatchSubTargets,
   199→  onSelectSubTarget,
   200→  onSelectPdb,
   201→}: {
   202→  batchId: string;
   203→  allEvals: Evaluation[];
   204→  evalBatches: EvalBatch[];
   205→  evalBatchSubTargets: Record<string, EvalBatchSubTarget[]>;
   206→  onSelectSubTarget: (uniprotId: string) => void;
   207→  onSelectPdb?: (pdbId: string) => void;
   208→}) {
   209→  const { locale } = useI18n();
   210→  const batch = evalBatches.find(b => b.batchId === batchId);
   211→  const subTargets = evalBatchSubTargets[batchId] || [];
   212→  const commonPdbIds = parseCommonPdbIds(batch?.commonPdbIds);
   213→
   214→  // Build a map of pdbId → list of UniProt IDs that have it (for "Shared By" column)
   215→  const pdbHolders = useMemo(() => {
   216→    const map: Record<string, string[]> = {};
   217→    for (const sub of subTargets) {
   218→      const evalData = allEvals.find(e => e.uniprotId === sub.uniprotId);
   219→      if (!evalData) continue;
   220→      for (const s of evalData.pdbStructures || []) {
   221→        if (!map[s.pdbId]) map[s.pdbId] = [];
   222→        if (!map[s.pdbId].includes(sub.uniprotId)) {
   223→          map[s.pdbId].push(sub.uniprotId);
   224→        }
   225→      }
   226→      for (const b of evalData.blastResults || []) {
   227→        if (!map[b.pdbId]) map[b.pdbId] = [];
   228→        if (!map[b.pdbId].includes(sub.uniprotId)) {
   229→          map[b.pdbId].push(sub.uniprotId);
   230→        }
   231→      }
   232→    }
   233→    return map;
   234→  }, [subTargets, allEvals]);
   235→
   236→  // Build rows for the common PDB table — EvalPdbTable-compatible EvalRow[].
   237→  // For each common pdbId, find the first eval that has it (as either a
   238→  // direct PDB structure or a BLAST hit) and surface its full RCSB fields.
   239→  // The "shared by" info is encoded into the row's `_source` and as a tooltip
   240→  // so EvalPdbTable can render the standard 9-column schema; we show the
   241→  // actual holder list in a small annotation chip below the table.
   242→  const commonPdbRows: EvalRow[] = useMemo(() => {
   243→    return commonPdbIds.map((pdbId): EvalRow => {
   244→      // Find the first eval that has this PDB to get full method/resolution/etc.
   245→      let found: EvalPdbStructure | null = null;
   246→      for (const sub of subTargets) {
   247→        const evalData = allEvals.find(e => e.uniprotId === sub.uniprotId);
   248→        if (!evalData) continue;
   249→        const struct = (evalData.pdbStructures || []).find(s => s.pdbId === pdbId);
   250→        if (struct) { found = struct; break; }
   251→        const blast = (evalData.blastResults || []).find(b => b.pdbId === pdbId);
   252→        if (blast) {
   253→          // Build a minimal EvalPdbStructure from a BLAST row so EvalPdbTable
   254→          // can render it. Missing fields are null.
   255→          found = {
   256→            id: -1,
   257→            uniprotId: evalData.uniprotId,
   258→            pdbId,
   259→            method: blast.method || null,
   260→            resolution: blast.resolution ?? null,
   261→            title: blast.title || blast.description || null,
   262→            depositionDate: null,
   263→            releaseDate: blast.releaseDate || null,
   264→            ligand: null,
   265→            ligandNames: null,
   266→            journal: null,
   267→            journalIf: blast.journalIf ?? null,
   268→            doi: null,
   269→            pubmedId: null,
   270→            organism: null,
   271→            authors: null,
   272→            isCryoem: false, isXray: false, isNmr: false,
   273→            ifTier: blast.ifTier || null,
   274→            chainId: null, unpStart: null, unpEnd: null,
   275→          } as EvalPdbStructure;
   276→          break;
   277→        }
   278→      }
   279→      if (!found) {
   280→        // Fallback: minimal row from pdbId only
   281→        found = {
   282→          id: -1, uniprotId: '', pdbId, method: null, resolution: null,
   283→          title: null, depositionDate: null, releaseDate: null,
   284→          ligand: null, ligandNames: null, journal: null, journalIf: null,
   285→          doi: null, pubmedId: null, organism: null, authors: null,
   286→          isCryoem: false, isXray: false, isNmr: false, ifTier: null,
   287→          chainId: null, unpStart: null, unpEnd: null,
   288→        } as EvalPdbStructure;
   289→      }
   290→      return {
   291→        ...found,
   292→        _type: 'structure' as const,
   293→        // Tag with the holder list (sub-targets sharing this PDB) so a
   294→        // downstream UI layer could show it. EvalPdbTable ignores unknown
   295→        // fields, so this is safe.
   296→        _sharedBy: pdbHolders[pdbId] || [],
   297→      } as EvalRow;
   298→    });
   299→  }, [commonPdbIds, subTargets, allEvals, pdbHolders]);
   300→
   301→  // Backward-compat: keep the old `rows` for the small "shared by" chip strip
   302→  // below the table — it just needs pdbId + holders per row.
   303→  const sharedByRows = useMemo(() => {
   304→    return commonPdbIds.map(pdbId => ({
   305→      pdbId,
   306→      holders: pdbHolders[pdbId] || [],
   307→    }));
   308→  }, [commonPdbIds, pdbHolders]);
   309→
   310→  if (!batch) {
   311→    return (
   312→      <div className="flex flex-col items-center justify-center py-16 text-claude-text-muted">
   313→        <Layers className="h-8 w-8 mb-2 opacity-40" />
   314→        <p className="text-xs">{locale === 'zh' ? '未找到批次' : 'Batch not found'}</p>
   315→      </div>
   316→    );
   317→  }
   318→
   319→  return (
   320→    <div className="flex flex-col h-full">
   321→      {/* Toolbar — batch title + stats */}
   322→      <div className="flex items-center gap-3 px-4 py-2.5 border-b border-claude-border dark:border-[#3d3832] bg-claude-surface dark:bg-[#242220] flex-shrink-0">
   323→        <div className="flex items-center gap-2 min-w-0">
   324→          <div className="h-8 w-8 rounded-lg bg-gradient-to-br from-violet-500/15 to-violet-500/5 border border-violet-500/20 flex items-center justify-center flex-shrink-0">
   325→            <Layers className="h-4 w-4 text-violet-600 dark:text-violet-300" />
   326→          </div>
   327→          <div className="min-w-0">
   328→            <div className="flex items-center gap-1.5">
   329→              <span className="text-sm font-semibold text-claude-text truncate">{batch.title || 'Batch'}</span>
   330→              <Badge variant="outline" className="text-[9px] font-semibold px-1.5 h-4 rounded bg-violet-50 dark:bg-violet-900/20 text-violet-700 dark:text-violet-300 border-violet-200 dark:border-violet-800/40 shrink-0">
   331→                {subTargets.length} targets
   332→              </Badge>
   333→            </div>
   334→            <div className="text-[10px] text-claude-text-muted font-mono">{batch.batchId}</div>
   335→          </div>
   336→        </div>
   337→        <div className="flex-1" />
   338→        <div className="flex items-center gap-3 text-[10px] text-claude-text-muted">
   339→          <span className="flex items-center gap-1">
   340→            <Box className="h-3 w-3" />
   341→            <span className="font-mono font-semibold text-claude-text-secondary">{commonPdbIds.length}</span> shared PDB
   342→          </span>
   343→        </div>
   344→      </div>
   345→
   346→      {/* Sub-target chips — clickable to open individual eval */}
   347→      <div className="flex items-center gap-1.5 px-4 py-2 border-b border-claude-border/50 dark:border-[#3d3832]/50 overflow-x-auto flex-shrink-0">
   348→        <span className="text-[9px] uppercase tracking-wider text-claude-text-muted font-semibold shrink-0 mr-1">{locale === 'zh' ? '靶点：' : 'Targets:'}</span>
   349→        {subTargets.length === 0 ? (
   350→          <span className="text-[10px] text-claude-text-muted italic">{locale === 'zh' ? '无子靶点' : 'No sub-targets'}</span>
   351→        ) : (
   352→          subTargets.map(sub => (
   353→            <button
   354→              key={sub.uniprotId}
   355→              onClick={() => onSelectSubTarget(sub.uniprotId)}
   356→              className="flex items-center gap-1 px-2 py-1 rounded-md text-[10px] font-mono font-medium bg-claude-border-light/60 dark:bg-[#2b2926]/60 hover:bg-claude-accent/10 hover:text-claude-accent text-claude-text-secondary transition-all border border-transparent hover:border-claude-accent/20 whitespace-nowrap"
   357→              title={`${sub.proteinName || sub.geneName || sub.uniprotId} · ${sub.pdbCount} PDB · ${sub.blastCount} BLAST`}
   358→            >
   359→              {sub.uniprotId}
   360→              <span className="text-[8px] text-claude-text-muted font-sans">{sub.pdbCount}P</span>
   361→            </button>
   362→          ))
   363→        )}
   364→      </div>
   365→
   366→      {/* Common PDB list — same 9-column schema as single eval EvalPdbTable */}
   367→      <div className="flex-1 min-h-0 overflow-hidden">
   368→        {commonPdbIds.length === 0 ? (
   369→          <div className="flex flex-col items-center justify-center py-16 text-claude-text-muted">
   370→            <Box className="h-8 w-8 mb-2 opacity-40" />
   371→            <p className="text-xs">No common PDB structures found between targets</p>
   372→            <p className="text-[10px] mt-1">Run the batch evaluation to detect shared structures.</p>
   373→          </div>
   374→        ) : (
   375→          <EvalPdbTable
   376→            rows={commonPdbRows}
   377→            loading={false}
   378→            compact={false}
   379→            onSelectRow={(row) => onSelectPdb?.(row.pdbId)}
   380→          />
   381→        )}
   382→      </div>
   383→
   384→      {/* Shared-by annotation strip — shows which sub-targets share each PDB
   385→          (kept because EvalPdbTable doesn't have a "shared by" column).
   386→          Maps each common pdbId → its holder list. */}
   387→      {sharedByRows.length > 0 && (
   388→        <div className="border-t border-claude-border dark:border-[#3d3832] bg-claude-surface dark:bg-[#242220] px-4 py-2 max-h-[120px] overflow-y-auto custom-scrollbar flex-shrink-0">
   389→          <div className="text-[9px] uppercase tracking-wider text-claude-text-muted font-semibold mb-1.5">
   390→            {locale === 'zh' ? '共享子靶点' : 'Shared By'}
   391→          </div>
   392→          <div className="flex flex-col gap-1">
   393→            {sharedByRows.map(({ pdbId, holders }) => (
   394→              <div key={pdbId} className="flex items-center gap-1.5 text-[10px]">
   395→                <span className="font-mono font-bold text-claude-accent shrink-0 w-[60px]">{pdbId}</span>
   396→                <div className="flex items-center gap-0.5 flex-wrap">
   397→                  {holders.length === 0 ? (
   398→                    <span className="text-claude-text-muted italic">—</span>
   399→                  ) : (
   400→                    <>
   401→                      {holders.slice(0, 6).map(h => (
   402→                        <span key={h} className="text-[9px] font-mono px-1 py-0.5 rounded bg-claude-border-light/60 dark:bg-[#2b2926]/60 text-claude-text-muted">
   403→                          {h}
   404→                        </span>
   405→                      ))}
   406→                      {holders.length > 6 && (
   407→                        <span className="text-[9px] text-claude-text-muted">+{holders.length - 6}</span>
   408→                      )}

   421→function getScoreColor(score: number): string {
   422→  if (score >= 80) return '#2d8f8f';

   428→// Parse Evaluation.scores JSON into a {key: {score, max}} map.
   429→// The DB stores scores as e.g. {"X-ray":{"score":8,"rating":"good","maxScore":10},...}
   430→function parseEvalScores(
   431→  scoresStr: string | null | undefined
   432→): Record<string, { score: number; max: number; rating?: string }> {
   433→  if (!scoresStr) return {};
   434→  try {
   435→    const parsed = JSON.parse(scoresStr);
   436→    if (typeof parsed === 'object' && parsed !== null) {
   437→      const result: Record<string, { score: number; max: number; rating?: string }> = {};
   438→      for (const [key, val] of Object.entries(parsed)) {
   439→        if (typeof val === 'number') {
   440→          result[key] = { score: val, max: 10 };
   441→        } else if (
   442→          typeof val === 'object' &&
   443→          val !== null &&
   444→          'score' in (val as Record<string, unknown>)
   445→        ) {
   446→          const v = val as {
   447→            score: number;
   448→            max?: number;
   449→            maxScore?: number;
   450→            rating?: string;
   451→          };
   452→          result[key] = {
   453→            score: v.score,
   454→            max: v.max ?? v.maxScore ?? 10,
   455→            rating: v.rating,
   456→          };
   457→        }
   458→      }
   459→      return result;
   460→    }
   461→  } catch {
   462→    /* ignore */
   463→  }
   464→  return {};
   465→}
   466→
   467→// Average each score category across all sub-target evals (for radar chart).
   468→function aggregateScores(
   469→  evals: (Evaluation | undefined)[]
   470→): Record<string, { score: number; max: number }> {
   471→  const sums: Record<string, { total: number; count: number; max: number }> = {};
   472→  for (const ev of evals) {
   473→    if (!ev?.scores) continue;
   474→    const parsed = parseEvalScores(ev.scores);
   475→    for (const [key, val] of Object.entries(parsed)) {
   476→      if (!sums[key]) sums[key] = { total: 0, count: 0, max: val.max || 10 };
   477→      sums[key].total += val.score;
   478→      sums[key].count += 1;
   479→      if (val.max > sums[key].max) sums[key].max = val.max;
   480→    }
   481→  }
   482→  const result: Record<string, { score: number; max: number }> = {};
   483→  for (const [key, val] of Object.entries(sums)) {
   484→    result[key] = {
   485→      score: val.count > 0 ? val.total / val.count : 0,
   486→      max: val.max,
   487→    };
   488→  }
   489→  return result;
   490→}
