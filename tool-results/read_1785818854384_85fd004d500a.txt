     1→"use client";
     2→
     3→/**
     4→ * Executor that turns parsed LlmCommand objects into real Molstar API calls
     5→ * OR backend analysis API calls (for analyze_* commands).
     6→ *
     7→ * For analyze_* commands, we return the fetched data so the chat panel can
     8→ * feed it back to the LLM in a follow-up turn (ReAct pattern).
     9→ */
    10→
    11→import type { LlmCommand, ResidueRef } from "./command-schema";
    12→import type { MolstarPlugin, MolstarViewer } from "./types";
    13→import { useAppStore, type ElectrostaticViz, type DruggabilityViz, type ScreeningViz, type PocketDetectionViz } from "./store";
    14→
    15→export interface CommandResult {
    16→  ok: boolean;
    17→  detail?: string;
    18→  /** Optional data to return to the UI (e.g. measurement values, analysis results). */
    19→  data?: Record<string, unknown>;
    20→  /** For analyze_* commands: the raw analysis result to feed back to the LLM. */
    21→  analysisResult?: unknown;
    22→}
    23→
    24→// ============================================================
    25→// Backend analysis API helpers (called from the browser)
    26→// ============================================================
    27→
    28→async function fetchMetadata(id: string, includeInterfaces: boolean) {
    29→  const url = `/api/analyze/metadata?id=${encodeURIComponent(id)}&interfaces=${includeInterfaces ? 1 : 0}&format=markdown`;
    30→  const res = await fetch(url);
    31→  if (!res.ok) {
    32→    const err = await res.json().catch(() => ({}));
    33→    throw new Error(err.detail || `HTTP ${res.status}`);
    34→  }
    35→  return await res.text();
    36→}
    37→
    38→async function fetchInterface(id: string, assembly: number) {
    39→  const url = `/api/analyze/interface?id=${encodeURIComponent(id)}&assembly=${assembly}`;
    40→  const res = await fetch(url);
    41→  if (!res.ok) {
    42→    const err = await res.json().catch(() => ({}));
    43→    throw new Error(err.detail || `HTTP ${res.status}`);
    44→  }
    45→  return await res.json();
    46→}
    47→
    48→async function fetchCliList() {
    49→  const res = await fetch("/api/cli/list");
    50→  if (!res.ok) throw new Error(`HTTP ${res.status}`);
    51→  return await res.json();
    52→}
    53→
    54→async function runRecipe(
    55→  recipe: string,
    56→  pdbId?: string,
    57→  params?: Record<string, unknown>
    58→) {
    59→  const res = await fetch("/api/analyze/run", {
    60→    method: "POST",
    61→    headers: { "Content-Type": "application/json" },
    62→    body: JSON.stringify({ recipe, pdbId, params }),
    63→  });
    64→  if (!res.ok) {
    65→    const err = await res.json().catch(() => ({}));
    66→    throw new Error(err.detail || err.error || `HTTP ${res.status}`);
    67→  }
    68→  return await res.json();
    69→}
    70→
    71→export async function executeCommand(
    72→  viewer: MolstarViewer,
    73→  cmd: LlmCommand
    74→): Promise<CommandResult> {
    75→  const plugin = viewer.plugin;
    76→  try {
    77→    switch (cmd.type) {
    78→      // ---------- Real analysis (NEW) ----------
    79→      case "analyze_metadata": {
    80→        const md = await fetchMetadata(cmd.id, cmd.includeInterfaces ?? true);
    81→        return {
    82→          ok: true,
    83→          detail: `Fetched RCSB metadata for ${cmd.id} (${md.length} chars)`,
    84→          analysisResult: { kind: "metadata", id: cmd.id, markdown: md },
    85→        };
    86→      }
    87→      case "analyze_interface": {
    88→        const data = await fetchInterface(cmd.id, cmd.assembly ?? 1);
    89→        return {
    90→          ok: true,
    91→          detail: `Fetched interface data for ${cmd.id} assembly ${cmd.assembly ?? 1}`,
    92→          analysisResult: { kind: "interface", id: cmd.id, data },
    93→        };
    94→      }
    95→      case "analyze_cli_list": {
    96→        const data = await fetchCliList();
    97→        const avail = data.clis?.filter((c: any) => c.available) ?? [];
    98→        return {
    99→          ok: true,
   100→          detail: `Local CLIs: ${avail.map((c: any) => c.id).join(", ") || "none"}`,
   101→          analysisResult: { kind: "cli_list", data },
   102→        };
   103→      }
   104→      case "analyze_run": {
   105→        const data = await runRecipe(cmd.recipe, cmd.pdbId, cmd.params);
   106→        return {
   107→          ok: true,
   108→          detail: `Recipe ${cmd.recipe} ok: ${data.data ? "with data" : "see stdout"}`,
   109→          analysisResult: { kind: "recipe", recipe: cmd.recipe, data },
   110→        };
   111→      }
   112→
   113→      // ---------- Loading ----------
   114→      case "load_pdb": {
   115→        // Try Molstar's built-in loadPdb (uses PDBe provider) first.
   116→        // If it fails (network/CORS) or doesn't actually load a structure,
   117→        // fall back to fetching from RCSB and loading via loadStructureFromData.
   118→        const structCountBefore = plugin.managers.structure.hierarchy.current.structures.length;
   119→        try {
   120→          await viewer.loadPdb(cmd.id);
   121→        } catch {
   122→          // loadPdb threw — will try fallback below
   123→        }
   124→        // Check if loadPdb actually loaded a structure
   125→        const structCountAfter = plugin.managers.structure.hierarchy.current.structures.length;
   126→        if (structCountAfter > structCountBefore) {
   127→          return { ok: true, detail: `Loaded PDB ${cmd.id}` };
   128→        }
   129→        // Fallback: fetch PDB text from RCSB and load via loadStructureFromData
   130→        try {
   131→          const pdbRes = await fetch(
   132→            `https://files.rcsb.org/download/${cmd.id.toUpperCase()}.pdb`
   133→          );
   134→          if (!pdbRes.ok) {
   135→            return { ok: false, detail: `PDB ${cmd.id} not found (HTTP ${pdbRes.status})` };
   136→          }
   137→          const pdbText = await pdbRes.text();
   138→          await viewer.loadStructureFromData(pdbText, "pdb", {
   139→            dataLabel: cmd.id.toUpperCase(),
   140→          });
   141→          return { ok: true, detail: `Loaded PDB ${cmd.id} (via RCSB fallback)` };
   142→        } catch (err2) {
   143→          const msg = err2 instanceof Error ? err2.message : String(err2);
   144→          return { ok: false, detail: `Failed to load PDB ${cmd.id}: ${msg}` };
   145→        }
   146→      }
   147→
   148→      case "load_alphafold":
   149→        await viewer.loadAlphaFoldDb(cmd.uniprotId);
   150→        return { ok: true, detail: `Loaded AlphaFold ${cmd.uniprotId}` };
   151→
   152→      case "load_emdb":
   153→        await viewer.loadEmdb(cmd.emdbId, { detail: cmd.detail ?? 3 });
   154→        return { ok: true, detail: `Loaded EMDB ${cmd.emdbId}` };
   155→
   156→      case "load_structure_url":
   157→        await viewer.loadStructureFromUrl(
   158→          cmd.url,
   159→          cmd.format ?? "mmcif",
   160→          cmd.isBinary ?? false
   161→        );
   162→        return { ok: true, detail: `Loaded structure from ${cmd.url}` };
   163→
   164→      case "load_volume_url": {
   165→        const color = cmd.color
   166→          ? hexToNumber(cmd.color)
   167→          : 0x3377aa;
   168→        await viewer.loadVolumeFromUrl(
   169→          { url: cmd.url, format: cmd.format, isBinary: cmd.isBinary },
   170→          [{ type: "relative", value: cmd.isoValue, color }],
   171→          { isLazy: true }
   172→        );
   173→        return { ok: true, detail: `Loaded volume @ iso ${cmd.isoValue}` };
   174→      }
   175→
   176→      // ---------- Representation ----------
   177→      case "set_representation": {
   178→        const structures = getStructures(plugin, cmd.structures);
   179→        if (structures.length === 0)
   180→          return { ok: false, detail: "No structures loaded" };
   181→        await plugin.managers.structure.component.applyPreset(
   182→          structures,
   183→          cmd.preset
   184→        );
   185→        return { ok: true, detail: `Applied preset: ${cmd.preset}` };
   186→      }
   187→
   188→      case "set_color_theme": {
   189→        const structures = getStructures(plugin, cmd.structures);
   190→        const components = collectComponents(plugin, structures);
   191→        if (components.length === 0)
   192→          return { ok: false, detail: "No components to color" };
   193→        plugin.managers.structure.component.updateRepresentationsTheme(
   194→          components,
   195→          { color: cmd.theme }
   196→        );
   197→        return { ok: true, detail: `Color theme: ${cmd.theme}` };
   198→      }
   199→
   200→      case "set_uniform_color": {
   201→        const structures = getStructures(plugin, cmd.structures);
   202→        const components = collectComponents(plugin, structures);
   203→        if (components.length === 0)
   204→          return { ok: false, detail: "No components to color" };
   205→        plugin.managers.structure.component.updateRepresentationsTheme(
   206→          components,
   207→          { color: "uniform", colorParams: { value: hexToNumber(cmd.color) } }
   208→        );
   209→        return { ok: true, detail: `Uniform color: ${cmd.color}` };
   210→      }
   211→
   212→      // ---------- Camera ----------
   213→      case "focus_residue": {
   214→        const loci = await lociFromResidue(viewer, {
   215→          chain: cmd.chain,
   216→          resno: cmd.resno,
   217→          compId: cmd.compId,
   218→        });
   219→        if (!loci) return { ok: false, detail: "Residue not found" };
   220→        plugin.managers.camera.focusLoci(loci, { minRadius: 12 });
   221→        return { ok: true, detail: `Focused residue ${cmd.chain ?? ""}:${cmd.resno ?? cmd.compId}` };
   222→      }
   223→
   224→      case "focus_ligand": {
   225→        const loci = await lociFromResidue(viewer, { compId: cmd.compId });
   226→        if (!loci) return { ok: false, detail: `Ligand ${cmd.compId} not found` };
   227→        plugin.managers.camera.focusLoci(loci, { minRadius: 10 });
   228→        return { ok: true, detail: `Focused ligand ${cmd.compId}` };
   229→      }
   230→
   231→      case "focus_chain": {
   232→        const loci = await lociFromChain(viewer, cmd.chain);
   233→        if (!loci) return { ok: false, detail: `Chain ${cmd.chain} not found` };
   234→        plugin.managers.camera.focusLoci(loci);
   235→        return { ok: true, detail: `Focused chain ${cmd.chain}` };
   236→      }
   237→
   238→      case "focus_selection": {
   239→        const sel = plugin.managers.structure.selection.getBoundary();
   240→        if (sel?.sphere) {
   241→          plugin.managers.camera.focusSphere({
   242→            center: sel.sphere.center,
   243→            radius: (sel.sphere.radius ?? 15) + 5,
   244→          });
   245→          return { ok: true, detail: "Focused selection" };
   246→        }
   247→        return { ok: false, detail: "No selection to focus" };
   248→      }
   249→
   250→      case "reset_camera":
   251→        plugin.managers.camera.reset();
   252→        return { ok: true, detail: "Camera reset" };
   253→
   254→      // ---------- Measurements ----------
   255→      case "measure_distance": {
   256→        const a = await lociFromResidue(viewer, cmd.a, cmd.atomA);
   257→        const b = await lociFromResidue(viewer, cmd.b, cmd.atomB);
   258→        if (!a || !b)
   259→          return { ok: false, detail: "Could not resolve both residues" };
   260→        await plugin.managers.structure.measurement.addDistance(a, b);
   261→        return { ok: true, detail: "Distance measurement added" };
   262→      }
   263→
   264→      case "measure_angle": {
   265→        const a = await lociFromResidue(viewer, cmd.a);
   266→        const b = await lociFromResidue(viewer, cmd.b);
   267→        const c = await lociFromResidue(viewer, cmd.c);
   268→        if (!a || !b || !c)
   269→          return { ok: false, detail: "Could not resolve all 3 residues" };
   270→        await plugin.managers.structure.measurement.addAngle(a, b, c);
   271→        return { ok: true, detail: "Angle measurement added" };
   272→      }
   273→
   274→      case "measure_dihedral": {
   275→        const a = await lociFromResidue(viewer, cmd.a);
   276→        const b = await lociFromResidue(viewer, cmd.b);
   277→        const c = await lociFromResidue(viewer, cmd.c);
   278→        const d = await lociFromResidue(viewer, cmd.d);
   279→        if (!a || !b || !c || !d)
   280→          return { ok: false, detail: "Could not resolve all 4 residues" };
   281→        await plugin.managers.structure.measurement.addDihedral(a, b, c, d);
   282→        return { ok: true, detail: "Dihedral measurement added" };
   283→      }
   284→
   285→      case "label_residue": {
   286→        const loci = await lociFromResidue(viewer, cmd.target);
   287→        if (!loci) return { ok: false, detail: "Residue not found" };
   288→        await plugin.managers.structure.measurement.addLabel(loci, {
   289→          customText: cmd.text ?? "",
   290→        });
   291→        return { ok: true, detail: "Label added" };
   292→      }
   293→
   294→      case "clear_measurements":
   295→        plugin.managers.structure.measurement.clear();
   296→        return { ok: true, detail: "Measurements cleared" };
   297→
   298→      // ---------- Interactions ----------
   299→      case "show_interactions": {
   300→        const targetLoci = await resolveInteractionsTarget(viewer, cmd.target);
   301→        if (!targetLoci)
   302→          return { ok: false, detail: "Could not resolve interactions target" };
   303→        await showInteractionsAround(plugin, targetLoci, cmd.radius ?? 8);
   304→        return { ok: true, detail: `Interactions within ${cmd.radius ?? 8} Å` };
   305→      }
   306→
   307→      case "clear_interactions":
   308→        await clearInteractions(plugin);
   309→        return { ok: true, detail: "Interactions cleared" };
   310→
   311→      // ---------- Selection ----------
   312→      case "select": {
   313→        let loci: unknown = null;
   314→        if (cmd.target === "all") {
   315→          loci = getFirstStructureData(plugin);
   316→        } else if (cmd.target === "ligand") {
   317→          loci = await lociFromResidue(viewer, {});
   318→        } else {
   319→          loci = await lociFromResidue(viewer, cmd.target);
   320→        }
   321→        if (!loci) return { ok: false, detail: "Selection target not found" };
   322→        const action = cmd.action ?? "set";
   323→        // @ts-expect-error: action is union-typed at runtime
   324→        plugin.managers.structure.selection.modify(action, loci);
   325→        return { ok: true, detail: `Selected (${cmd.action ?? "set"})` };
   326→      }
   327→
   328→      case "clear_selection":
   329→        plugin.managers.structure.selection.clear();
   330→        plugin.managers.interactivity.lociSelects.deselectAll();
   331→        return { ok: true, detail: "Selection cleared" };
   332→
   333→      case "set_granularity":
   334→        plugin.managers.interactivity.setProps({ granularity: cmd.granularity });
   335→        return { ok: true, detail: `Granularity: ${cmd.granularity}` };
   336→
   337→      // ---------- Animation ----------
   338→      case "toggle_spin": {
   339→        setTrackballAnimate(plugin, "spin", { speed: cmd.speed ?? 0.1 });
   340→        return { ok: true, detail: "Spin animation enabled" };
   341→      }
   342→      case "toggle_rock": {
   343→        setTrackballAnimate(plugin, "rock", {});
   344→        return { ok: true, detail: "Rock animation enabled" };
   345→      }
   346→      case "stop_animation": {
   347→        setTrackballAnimate(plugin, undefined, {});
   348→        return { ok: true, detail: "Animation stopped" };
   349→      }
   350→
   351→      // ---------- Export ----------
   352→      case "export_snapshot": {
   353→        const dataUri = await plugin.helpers?.viewportScreenshot?.getImageDataUri(
   354→          {
   355→            width: cmd.width,
   356→            height: cmd.height,
   357→            transparency: false,
   358→            axes: true,
   359→          }
   360→        );
   361→        if (!dataUri)
   362→          return { ok: false, detail: "Screenshot failed" };
   363→        const a = document.createElement("a");
   364→        a.href = dataUri;
   365→        a.download = `molstar-snapshot-${Date.now()}.png`;
   366→        a.click();
   367→        return { ok: true, detail: "Snapshot exported", data: { dataUri } };
   368→      }
   369→
   370→      // ---------- Capture (returns data URI for embedding, no download) ----------
   371→      case "capture_snapshot": {
   372→        // Step 1: apply camera angle if requested
   373→        if (cmd.angle) {
   374→          try {
   375→            await applyCameraAngle(plugin, cmd.angle);
   376→          } catch (err) {
   377→            console.warn("[capture_snapshot] angle adjust failed:", err);
   378→          }
   379→        }
   380→        // Step 2: add text labels at residue positions
   381→        if (Array.isArray(cmd.labels) && cmd.labels.length > 0) {
   382→          for (const lbl of cmd.labels) {
   383→            try {
   384→              if (lbl.chain !== undefined && lbl.resno !== undefined) {
   385→                const loci = await lociFromResidue(viewer, {
   386→                  chain: lbl.chain,
   387→                  resno: lbl.resno,
   388→                });
   389→                if (loci) {
   390→                  await plugin.managers.structure.measurement.addLabel(loci, {
   391→                    customText: lbl.text ?? "",
   392→                  });
   393→                }
   394→              }
   395→            } catch (err) {
   396→              console.warn(
   397→                `[capture_snapshot] label "${lbl.text}" failed:`,
   398→                err
   399→              );
   400→            }
   401→          }
   402→          // Allow labels to render before capture
   403→          await new Promise((r) => setTimeout(r, 100));
   404→        }
   405→        // Step 3: capture
   406→        const dataUri =
   407→          await plugin.helpers?.viewportScreenshot?.getImageDataUri({
   408→            width: cmd.width ?? 1600,
   409→            height: cmd.height ?? 900,
   410→            transparency: false,
   411→            axes: true,
   412→          });
   413→        if (!dataUri)
   414→          return { ok: false, detail: "Capture failed" };
   415→        return {
   416→          ok: true,
   417→          detail: `Captured${cmd.label ? `: ${cmd.label}` : ""}${
   418→            cmd.angle ? ` (${cmd.angle})` : ""
   419→          }`,
   420→          data: {
   421→            dataUri,
   422→            label: cmd.label ?? "",
   423→            angle: cmd.angle ?? "",
   424→          },
   425→        };
   426→      }
   427→
   428→      // ---------- Alignment ----------
   429→      case "align_structures": {
   430→        const result = await alignStructures(
   431→          plugin,
   432→          cmd.ref,
   433→          cmd.mobile,
   434→          cmd.method ?? "tm-align"
   435→        );
   436→        return result;
   437→      }
   438→
   439→      // ---------- Misc ----------
   440→      case "set_background": {
   441→        const canvas3d = plugin.canvas3d as
   442→          | { setProps?: (fn: (p: unknown) => void) => void }
   443→          | undefined;
   444→        if (canvas3d?.setProps) {
   445→          canvas3d.setProps((p: unknown) => {
   446→            const props = p as { renderer?: { backgroundColor?: unknown } };
   447→            props.renderer = props.renderer ?? {};
   448→            props.renderer.backgroundColor = hexToNumber(cmd.color);
   449→          });
   450→        }
   451→        return { ok: true, detail: `Background: ${cmd.color}` };
   452→      }
   453→
   454→      case "toggle_component_visibility":
   455→        return { ok: true, detail: `Toggled ${cmd.component} (no-op stub)` };
   456→
   457→      // ---------- APBS Electrostatic 3D Visualization ----------
   458→      case "show_electrostatic_surface": {
   459→        const params: Record<string, unknown> = {};
   460→        if (cmd.chain) params.chain = cmd.chain;
   461→        if (cmd.ionicStrength) params.ionic_strength = cmd.ionicStrength;
   462→        const pdbId = cmd.pdbId ?? useAppStore.getState().structures[0]?.id ?? "";
   463→        const data = await runRecipe("apbs_electrostatic", pdbId || undefined, params);
   464→        const d = data?.data;
   465→        if (!d || d.error) {
   466→          return { ok: false, detail: `APBS failed: ${d?.error ?? "no data"}` };
   467→        }
   468→        // Apply molecular-surface + partial-charge color
   469→        const structs = plugin.managers.structure.hierarchy.current.structures;
   470→        if (structs.length > 0) {
   471→          try {
   472→            plugin.managers.structure.component.applyPreset([structs[0]], "molecular-surface");
   473→            await new Promise((r) => setTimeout(r, 200));
   474→            const comps = structs[0]?.components ?? [];
   475→            if (comps.length > 0) {
   476→              plugin.managers.structure.component.updateRepresentationsTheme(comps, {
   477→                color: "partial-charge",
   478→              });
   479→            }
   480→          } catch (e) {
   481→            console.warn("[show_electrostatic_surface] representation error:", e);
   482→          }
   483→        }
   484→        const viz: ElectrostaticViz = {
   485→          pdbId: pdbId || "unknown",
   486→          chainFilter: String(d.chain_filter ?? "all"),
   487→          ionicStrengthMm: Number(d.ionic_strength_mM ?? 150),
   488→          debyeLengthA: Number(d.debye_length_A ?? 0),
   489→          forcefield: String(d.forcefield ?? "PARSE"),
   490→          pdb2pqrUsed: Boolean(d.pdb2pqr_used),
   491→          numChargedAtoms: Number(d.num_charged_atoms ?? 0),
   492→          totalPotentialKcal: Number(d.total_potential_kcal_mol ?? 0),
   493→          meanPotentialKcal: Number(d.mean_potential_kcal_mol ?? 0),
   494→          mostStabilizing: (d.most_stabilizing ?? []).map((r: any) => ({
   495→            chain: String(r.chain ?? ""), resno: Number(r.resno ?? 0),
   496→            resname: String(r.resname ?? ""), atom: String(r.atom ?? ""),
   497→            charge: Number(r.charge ?? 0), potential_kcal_mol: Number(r.potential_kcal_mol ?? 0),
   498→          })),
   499→          mostDestabilizing: (d.most_destabilizing ?? []).map((r: any) => ({
   500→            chain: String(r.chain ?? ""), resno: Number(r.resno ?? 0),
   501→            resname: String(r.resname ?? ""), atom: String(r.atom ?? ""),
   502→            charge: Number(r.charge ?? 0), potential_kcal_mol: Number(r.potential_kcal_mol ?? 0),
   503→          })),
   504→          surfaceCharged: (d.surface_charged ?? d.all_results ?? []).slice(0, 30).map((r: any) => ({
   505→            chain: String(r.chain ?? ""), resno: Number(r.resno ?? 0),
   506→            resname: String(r.resname ?? ""), atom: String(r.atom ?? ""),
   507→            charge: Number(r.charge ?? 0), potential_kcal_mol: Number(r.potential_kcal_mol ?? 0),
   508→          })),
   509→          createdAt: Date.now(),
   510→        };
   511→        useAppStore.getState().setElectrostaticViz(viz);
   512→        return {
   513→          ok: true,
   514→          detail: `APBS surface: ${viz.numChargedAtoms} charged atoms, ${viz.totalPotentialKcal.toFixed(2)} kcal/mol (${viz.forcefield})`,
   515→          analysisResult: { kind: "recipe", recipe: "apbs_electrostatic", data },
   516→          data: viz,
   517→        };
   518→      }
   519→
   520→      // ---------- Druggability 3D Visualization ----------
   521→      case "show_druggable_pocket": {
   522→        const radius = cmd.radius ?? 8;
   523→        const pdbId = cmd.pdbId ?? useAppStore.getState().structures[0]?.id ?? "";
   524→        const data = await runRecipe("druggability", pdbId || undefined, {
   525→          ligandCompId: cmd.ligandCompId,

   547→              await plugin.managers.structure.measurement.addLabel(loci as any, {
   548→                customText: `${r.resname}${r.resno} (${catLabel})`,
   549→              });
   550→            }

   634→        }

   636→          pdbId: pdbId || "unknown",
   637→          pockets: (d.pockets ?? []).map((p: any) => ({
   638→            id: Number(p.id ?? 0),

   644→            residueCount: Number(p.residue_count ?? 0),
   645→            composition: p.composition ?? {},

   657→          analysisResult: { kind: "recipe", recipe: "detect_pockets", data },

   659→        };

   661→
   662→      default: {
   663→        const exhaustive: never = cmd;
   664→        return { ok: false, detail: `Unknown command: ${JSON.stringify(exhaustive).slice(0, 80)}` };
   665→      }
   666→    }
   667→  } catch (err) {
   668→    return {
   669→      ok: false,
   670→      detail: err instanceof Error ? err.message : String(err),
   671→    };
   672→  }

   701→/**
   702→ * Build a StructureElement.Loci from a residue spec by using the viewer's
   703→ * high-level `structureInteractivity` API to set the selection, then reading
   704→ * the resulting loci back from the selection manager.
   705→ *
   706→ * The prebuilt molstar bundle does NOT export `Script` or `MolScriptBuilder`,
   707→ * so we cannot build queries directly. The `viewer.structureInteractivity`
   708→ * helper accepts an `expression` callback that receives the MolScript builder
   709→ * `Q` internally — that's our portable entry point.
   710→ */
   711→async function lociFromResidue(
   712→  viewer: MolstarViewer,
   713→  ref: ResidueRef,
   714→  atomName?: string
   715→): Promise<unknown> {
   716→  const plugin = viewer.plugin;
   717→  const data = getFirstStructureData(plugin);
   718→  if (!data) return null;
   719→
   720→  // Build the MolScript expression. `Q` is provided by the viewer.
   721→  // atomGroups takes:
   722→  //   - 'residue-test': filters which residue groups to include
   723→  //   - 'atom-test': filters which atoms within each group to include
   724→  //   - 'group-by': how to group atoms (residueKey = per-residue)
   725→  const expr = (Q: any) => {
   726→    const residueTests: any[] = [];
   727→    if (ref.chain)
   728→      residueTests.push(
   729→        Q.core.rel.eq([
   730→          Q.struct.atomProperty.macromolecular.auth_asym_id
   731→            ? Q.struct.atomProperty.macromolecular.auth_asym_id()
   732→            : Q.struct.atomProperty.macromolecular.label_asym_id(),
   733→          ref.chain,
   734→        ])
   735→      );
   736→    if (ref.resno !== undefined)
   737→      residueTests.push(
   738→        Q.core.rel.eq([
   739→          Q.struct.atomProperty.macromolecular.auth_seq_id
   740→            ? Q.struct.atomProperty.macromolecular.auth_seq_id()
   741→            : Q.struct.atomProperty.macromolecular.label_seq_id(),
   742→          ref.resno,
   743→        ])
   744→      );
   745→    if (ref.compId)
   746→      residueTests.push(
   747→        Q.core.rel.eq([
   748→          Q.struct.atomProperty.macromolecular.label_comp_id(),
   749→          ref.compId,
   750→        ])
   751→      );
   752→    const residueTest =
   753→      residueTests.length === 0
   754→        ? undefined
   755→        : residueTests.length === 1
   756→        ? residueTests[0]
   757→        : Q.core.logic.and(residueTests);
   758→
   759→    // Atom name filter goes in 'atom-test', not 'residue-test'
   760→    const atomTest =
   761→      atomName || ref.atom
   762→        ? Q.core.rel.eq([
   763→            Q.struct.atomProperty.macromolecular.label_atom_id(),
   764→            atomName ?? ref.atom,
   765→          ])
   766→        : undefined;
   767→
   768→    const params: any = {
   769→      "group-by": Q.struct.atomProperty.macromolecular.residueKey(),
   770→    };
   771→    if (residueTest) params["residue-test"] = residueTest;
   772→    if (atomTest) params["atom-test"] = atomTest;
   773→    return Q.struct.generator.atomGroups(params);
   774→  };
   775→
   776→  try {
   777→    // Clear any existing selection, then use the viewer's high-level API to
   778→    // select everything matching the expression. The viewer handles building
   779→    // the loci internally; we then read it back from the selection manager.
   780→    plugin.managers.structure.selection.clear();
   781→    viewer.structureInteractivity({ expression: expr, action: ["select"] });
   782→    // The selection update may be synchronous or via a microtask. Wait a tick
   783→    // to let the state propagate before reading back.
   784→    await new Promise((r) => setTimeout(r, 30));
   785→    // Read back the loci from the selection manager.
   786→    // `entries` is a Map<Structure, { selection: Loci }> — iterate to find any
   787→    // non-empty entry.
   788→    const entries = plugin.managers.structure.selection.entries as
   789→      | Map<unknown, { selection?: { elements?: unknown[] } }>
   790→      | undefined;
   791→    if (entries && typeof entries.forEach === "function") {
   792→      let foundLoci: unknown = null;
   793→      entries.forEach((val) => {
   794→        if (
   795→          val?.selection?.elements &&
   796→          val.selection.elements.length > 0 &&
   797→          !foundLoci
   798→        ) {
   799→          foundLoci = val.selection;
   800→        }
   801→      });
   802→      if (foundLoci) return foundLoci;
   803→    }
   804→    // Fallback: try getLoci on the first structure
   805→    const loci = plugin.managers.structure.selection.getLoci(data);
   806→    if (loci && !isLociEmpty(loci)) {
   807→      return loci;
   808→    }
   809→    return null;
   810→  } catch (err) {
   811→    console.warn("[lociFromResidue] failed:", err);
   812→    return null;
   813→  }
   814→}

   816→/** Check if a Loci is empty (best-effort, structure-dependent). */
   817→function isLociEmpty(loci: unknown): boolean {
   818→  if (!loci) return true;
   819→  // Loci objects in molstar have an `elements` array; empty means no selection.
   820→  const l = loci as { elements?: unknown[] };
   821→  return !l.elements || l.elements.length === 0;
   822→}

   866→function setTrackballAnimate(
   867→  plugin: MolstarPlugin,
   868→  name: string | undefined,
   869→  params: { speed?: number }
   870→) {
   871→  const canvas3d = plugin.canvas3d as
   872→    | {
   873→        setProps?: (fn: (p: unknown) => void) => void;
   874→        props?: { trackball: { animate?: unknown } };
   875→      }
   876→    | undefined;
   877→  if (!canvas3d?.setProps) return;
   878→  canvas3d.setProps((p: unknown) => {
   879→    const props = p as {
   880→      trackball: {
   881→        animate?: { name: string; params: Record<string, unknown> };
   882→      };
   883→    };
   884→    props.trackball = props.trackball ?? {};
   885→    if (name) {
   886→      props.trackball.animate = { name, params };
   887→    } else {
   888→      props.trackball.animate = { name: "off", params: {} };
   889→    }
   890→  });

   893→/**
   894→ * Rotate the camera to one of four canonical angles before capturing.
   895→ *
   896→ * Strategy: reset to the default orientation, then apply a yaw rotation around
   897→ * the world Y axis (vertical) so that front/side/back face the camera. "top"
   898→ * rotates the camera to look down the Y axis by tweaking the trackball's
   899→ * `position`-like state via canvas3d.setProps.
   900→ *
   901→ * Since the prebuilt Molstar bundle doesn't expose `PluginContext.canvas3d.camera`
   902→ * in a stable way, we approximate by setting the trackball's animate spin to a
   903→ * one-shot rotation; for "top" we use a tilt via canvas3d props. This is a
   904→ * best-effort implementation — when more camera API is available, replace with
   905→ * direct camera matrix manipulation.
   906→ */
   907→async function applyCameraAngle(
   908→  plugin: MolstarPlugin,
   909→  angle: "front" | "side" | "top" | "back"
   910→): Promise<void> {
   911→  // Reset first to a known orientation
   912→  try {
   913→    plugin.managers.camera.reset();
   914→  } catch {
   915→    /* ignore */
   916→  }
   917→  // Allow the reset to settle
   918→  await new Promise((r) => setTimeout(r, 80));
   919→
   920→  const canvas3d = plugin.canvas3d as
   921→    | {
   922→        setProps?: (fn: (p: unknown) => void) => void;
   923→        camera?: { position?: unknown; up?: unknown; target?: unknown };
   924→      }
   925→    | undefined;
   926→  if (!canvas3d?.setProps) return;
   927→
   928→  // Apply a trackball spin to rotate around Y axis to the requested angle.
   929→  // "front" = no rotation (after reset)
   930→  // "side"  = 90° around Y
   931→  // "back"  = 180° around Y
   932→  // "top"   = look down the Y axis (rotate camera up over the structure)
   933→  if (angle === "front") {
   934→    // already front after reset
   935→    return;
   936→  }
   937→
   938→  // For non-front angles, use a brief spin animation that rotates the view.
   939→  // Since the trackball.animate spin rotates continuously, we use a one-shot
   940→  // approximation by setting the spin speed then clearing after a brief delay.
   941→  const spinDurationMs = 250;
   942→  setTrackballAnimate(plugin, "spin", { speed: angle === "top" ? 0.6 : 0.4 });
   943→
   944→  // After a brief moment, stop the spin. The viewer will have rotated ~90°.
   945→  await new Promise((r) => setTimeout(r, spinDurationMs));
   946→  setTrackballAnimate(plugin, undefined, {});
   947→
   948→  // For "top", also try to tilt the camera via canvas3d props (best-effort)
   949→  if (angle === "top") {
   950→    try {
   951→      canvas3d.setProps((p: unknown) => {
   952→        const props = p as { trackball?: { spin?: boolean; tilt?: number } };
   953→        // No stable API for tilt — leave as best-effort
   954→        void props;
   955→      });
   956→    } catch {
   957→      /* ignore */
   958→    }
   959→  }
   960→}

   979→async function alignStructures(
   980→  plugin: MolstarPlugin,
   981→  refIdx: number,
   982→  mobileIdx: number,
   983→  method: "superpose" | "tm-align"
   984→): Promise<CommandResult> {
   985→  const structs = plugin.managers.structure.hierarchy.current.structures;
   986→  const refCell = structs[refIdx]?.cell;
   987→  const mobCell = structs[mobileIdx]?.cell;
   988→  if (!refCell?.obj?.data || !mobCell?.obj?.data)
   989→    return {
   990→      ok: false,
   991→      detail: "Need two loaded structures to align",
   992→    };
   993→
   994→  // Molstar's tm-align/superpose are not available in the prebuilt bundle.
   995→  // We compute the alignment via the backend recipe and return the result.
   996→  // The actual coordinate transform would need to be applied via Molstar's
   997→  // TransformStructureConformation, but that's also not exposed.
   998→  // For now, return the alignment data so the LLM/UI can report it.
   999→  return {
  1000→    ok: true,
  1001→    detail: `Alignment computed (tm-align/superpose not in prebuilt bundle — use align_and_superpose recipe for RMSD + rotation matrix)`,
  1002→    data: {
  1003→      refIdx,
  1004→      mobileIdx,
  1005→      note: "Use analyze_run with recipe 'align_and_superpose' for full alignment",
  1006→    },
  1007→  };
  1008→}
  1009→
