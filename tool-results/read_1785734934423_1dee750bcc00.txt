   200→function StructuresTab() {
   201→  const structures = useAppStore((s) => s.structures);
   202→  const removeStructure = useAppStore((s) => s.removeStructure);
   203→  const clearStructures = useAppStore((s) => s.clearStructures);
   204→  const activeStructureId = useAppStore((s) => s.activeStructureId);
   205→  const setActiveStructure = useAppStore((s) => s.setActiveStructure);
   206→  const updateStructureStyle = useAppStore((s) => s.updateStructureStyle);
   207→  const viewer = useAppStore((s) => s.viewer);
   208→  const toast = useAppStore((s) => s.toast);
   209→  const [hiddenSet, setHiddenSet] = useState<Set<string>>(new Set());
   210→  const [openId, setOpenId] = useState<string | null>(null);
   211→  const [aligning, setAligning] = useState(false);
   212→
   213→  useEffect(() => {
   214→    // eslint-disable-next-line react-hooks/set-state-in-effect
   215→    if (activeStructureId) setOpenId(activeStructureId);
   216→  }, [activeStructureId]);
   217→
   218→  const toggleVisibility = async (id: string) => {
   219→    if (!viewer) return;
   220→    try {
   221→      const plugin = viewer.plugin;
   222→      const structs = plugin.managers.structure.hierarchy.current.structures;
   223→      const idx = structures.findIndex((s) => s.id === id);
   224→      if (idx < 0 || idx >= structs.length) return;
   225→      plugin.managers.structure.hierarchy.toggleVisibility([structs[idx]]);
   226→      setHiddenSet((prev) => {
   227→        const next = new Set(prev);
   228→        if (next.has(id)) next.delete(id);
   229→        else next.add(id);
   230→        return next;
   231→      });
   232→    } catch (err) {
   233→      toast(`Toggle failed: ${err}`, "error");
   234→    }
   235→  };
   236→
   237→  const closeStructure = async (id: string) => {
   238→    if (!viewer) return;
   239→    try {
   240→      const plugin = viewer.plugin;
   241→      const structs = plugin.managers.structure.hierarchy.current.structures;
   242→      const idx = structures.findIndex((s) => s.id === id);
   243→      if (idx >= 0 && idx < structs.length) {
   244→        plugin.managers.structure.hierarchy.remove(structs[idx]);
   245→      }
   246→      removeStructure(id);
   247→      toast(`Removed ${id}`, "info");
   248→    } catch (err) {
   249→      toast(`Remove failed: ${err}`, "error");
   250→    }
   251→  };
   252→
   253→  if (structures.length === 0) {
   254→    return (
   255→      <div className="sa-empty-state p-4">
   256→        <Layers className="h-8 w-8 text-claude-text-muted" />
   257→        <p className="text-xs">No structures loaded</p>
   258→        <p className="text-[10px] text-claude-text-muted">
   259→          Use the toolbar above to load a PDB / AlphaFold / EMDB ID or upload a
   260→          file.
   261→        </p>
   262→      </div>
   263→    );
   264→  }
   265→
   266→  return (
   267→    <div className="space-y-1.5 p-2">
   268→      {structures.map((s, i) => {
   269→        const isActive = activeStructureId === s.id || (!activeStructureId && i === 0);
   270→        const isOpen = openId === s.id;
   271→        const isHidden = hiddenSet.has(s.id);
   272→        return (
   273→          <div key={s.id}>
   274→            <div
   275→              className={`sa-struct-item ${isActive ? "sa-struct-active" : ""}`}
   276→              onClick={() => {
   277→                setActiveStructure(s.id);
   278→                setOpenId(isOpen ? null : s.id);
   279→              }}
   280→            >
   281→              <span
   282→                className="h-2 w-2 shrink-0 rounded-full"
   283→                style={{ backgroundColor: s.color ?? "#c96442" }}
   284→              />
   285→              <span className="flex-1 truncate font-mono text-[11px] font-medium">
   286→                {s.label}
   287→              </span>
   288→              {s.metadata?.method && (
   289→                <Badge variant="outline" className="px-1 py-0 text-[8px]">
   290→                  {s.metadata.method}
   291→                </Badge>
   292→              )}
   293→              <button
   294→                onClick={(e) => {
   295→                  e.stopPropagation();
   296→                  toggleVisibility(s.id);
   297→                }}
   298→                className="text-claude-text-muted hover:text-claude-text"
   299→                title={isHidden ? "Show" : "Hide"}
   300→              >
   301→                {isHidden ? (
   302→                  <EyeOff className="h-3 w-3" />
   303→                ) : (
   304→                  <Eye className="h-3 w-3" />
   305→                )}
   306→              </button>
   307→              <button
   308→                onClick={(e) => {
   309→                  e.stopPropagation();
   310→                  closeStructure(s.id);
   311→                }}
   312→                className="text-claude-text-muted hover:text-destructive"
   313→                title="Remove"
   314→              >
   315→                <X className="h-3 w-3" />
   316→              </button>
   317→            </div>
   318→
   319→            {/* Inline expanded controls */}
   320→            {isOpen && (
   321→              <div className="mt-1 space-y-2 rounded-md border border-claude-border bg-claude-bg p-2">
   322→                {s.metadata && (
   323→                  <div className="grid grid-cols-2 gap-1 text-[9px]">
   324→                    {s.metadata.title && (
   325→                      <div className="col-span-2 truncate text-claude-text-secondary">
   326→                        <span className="font-medium">Title:</span>{" "}
   327→                        {s.metadata.title}
   328→                      </div>
   329→                    )}
   330→                    {s.metadata.chains && (
   331→                      <div className="text-claude-text-secondary">
   332→                        <span className="font-medium">Chains:</span>{" "}
   333→                        {s.metadata.chains.join(", ")}
   334→                      </div>
   335→                    )}
   336→                    {s.metadata.numResidues != null && (
   337→                      <div className="text-claude-text-secondary">
   338→                        <span className="font-medium">Residues:</span>{" "}
   339→                        {s.metadata.numResidues}
   340→                      </div>
   341→                    )}
   342→                    {s.metadata.numAtoms != null && (
   343→                      <div className="text-claude-text-secondary">
   344→                        <span className="font-medium">Atoms:</span>{" "}
   345→                        {s.metadata.numAtoms}
   346→                      </div>
   347→                    )}
   348→                    {s.metadata.resolution != null && (
   349→                      <div className="text-claude-text-secondary">
   350→                        <span className="font-medium">Res:</span>{" "}
   351→                        {s.metadata.resolution} Å
   352→                      </div>
   353→                    )}
   354→                  </div>
   355→                )}
   356→                {/* Representation */}
   357→                <div>
   358→                  <Label className="mb-1 block text-[9px] text-claude-text-secondary">
   359→                    Representation
   360→                  </Label>
   361→                  <Select
   362→                    value={s.style?.representation ?? "cartoon"}
   363→                    onValueChange={(v) =>
   364→                      updateStructureStyle(s.id, {
   365→                        representation: v as LoadedStructure["style"] extends infer T
   366→                          ? T extends { representation: infer R }
   367→                            ? R
   368→                            : never
   369→                          : never,
   370→                      })
   371→                    }
   372→                  >
   373→                    <SelectTrigger className="h-7 text-[10px]">
   374→                      <SelectValue />
   375→                    </SelectTrigger>
   376→                    <SelectContent>
   377→                      {["cartoon", "stick", "line", "sphere", "surface"].map(
   378→                        (r) => (
   379→                          <SelectItem key={r} value={r} className="text-[10px]">
   380→                            {r}
   381→                          </SelectItem>
   382→                        )
   383→                      )}
   384→                    </SelectContent>
   385→                  </Select>
   386→                </div>
   387→                {/* Color scheme */}
   388→                <div>
   389→                  <Label className="mb-1 block text-[9px] text-claude-text-secondary">
   390→                    Color scheme
   391→                  </Label>
   392→                  <Select
   393→                    value={s.style?.colorScheme ?? "spectrum"}
   394→                    onValueChange={(v) =>
   395→                      updateStructureStyle(s.id, {
   396→                        colorScheme: v as LoadedStructure["style"] extends infer T
   397→                          ? T extends { colorScheme: infer R }
   398→                            ? R
   399→                            : never
   400→                          : never,
   401→                      })
   402→                    }
   403→                  >
   404→                    <SelectTrigger className="h-7 text-[10px]">
   405→                      <SelectValue />
   406→                    </SelectTrigger>
   407→                    <SelectContent>
   408→                      {[
   409→                        "chain",
   410→                        "element",
   411→                        "secondary",
   412→                        "single",
   413→                        "spectrum",
   414→                        "bfactor",
   415→                        "residue",
   416→                        "charge",
   417→                      ].map((c) => (
   418→                        <SelectItem key={c} value={c} className="text-[10px]">
   419→                          {c}
   420→                        </SelectItem>
   421→                      ))}
   422→                    </SelectContent>
   423→                  </Select>
   424→                </div>
   425→                {s.alignRmsd != null && (
   426→                  <div className="rounded bg-claude-accent-light px-1.5 py-1 text-[9px] text-claude-accent">

   432→              </div>
   433→            )}
   434→          </div>
   435→        );
   436→      })}
   437→
   438→      {structures.length > 1 && (
   439→        <Button
   440→          size="sm"
   441→          variant="ghost"
   442→          className="mt-2 h-7 w-full text-[10px] text-claude-text-muted hover:text-destructive"
   443→          onClick={() => clearStructures()}
   444→        >
   445→          <Trash2 className="h-3 w-3" />
   446→          Clear all

   449→
   450→      {/* Sequence viewer — shown when a structure with PDB text is loaded */}

   466→        <StructureAlignmentPanel />
   467→      </div>
   468→    </div>
   469→  );
   470→}
