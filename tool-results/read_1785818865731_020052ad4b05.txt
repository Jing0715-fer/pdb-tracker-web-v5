     1→// Pure utilities for parsing PDB text and computing structure superposition.
     2→// All functions are client-safe (no Node APIs) so they can run in the browser.
     3→
     4→export interface CAAtom {
     5→  chain: string;
     6→  resSeq: number; // residue sequence number
     7→  resName: string;
     8→  x: number;
     9→  y: number;
    10→  z: number;
    11→}
    12→
    13→export interface ParsedPdb {
    14→  ca: CAAtom[];
    15→  chains: string[];
    16→  numAtoms: number;
    17→  numResidues: number;
    18→  title: string;
    19→}
    20→
    21→/**
    22→ * Parse PDB text. Extracts TITLE, all ATOM/HETATM records count, unique chains,
    23→ * residue count, and the C-alpha atoms (used for alignment).
    24→ */
    25→export function parsePdb(pdb: string): ParsedPdb {
    26→  const lines = pdb.split(/\r?\n/);
    27→  const ca: CAAtom[] = [];
    28→  const chainSet = new Set<string>();
    29→  const residueSet = new Set<string>();
    30→  let numAtoms = 0;
    31→  let title = "";
    32→
    33→  for (const line of lines) {
    34→    const rec = line.substring(0, 6).trim();
    35→    if (rec === "TITLE") {
    36→      // PDB TITLE columns 11-80 may be continued; take the text portion.
    37→      const text = line.substring(10).trim();
    38→      if (text) title = title ? `${title} ${text}` : text;
    39→    } else if (rec === "ATOM" || rec === "HETATM") {
    40→      numAtoms++;
    41→      const chain = line.substring(21, 22).trim() || " ";
    42→      chainSet.add(chain);
    43→      const atomName = line.substring(12, 16).trim();
    44→      const resName = line.substring(17, 20).trim();
    45→      const resSeq = parseInt(line.substring(22, 26), 10);
    46→      // Count polymer residues from ATOM records only (exclude HETATM waters/ligands).
    47→      if (!Number.isNaN(resSeq) && rec === "ATOM") residueSet.add(`${chain}:${resSeq}`);
    48→      if (atomName === "CA" && rec === "ATOM") {
    49→        const x = parseFloat(line.substring(30, 38));
    50→        const y = parseFloat(line.substring(38, 46));
    51→        const z = parseFloat(line.substring(46, 54));
    52→        if (!Number.isNaN(x) && !Number.isNaN(y) && !Number.isNaN(z)) {
    53→          ca.push({ chain, resSeq, resName, x, y, z });
    54→        }
    55→      }
    56→    }
    57→  }
    58→
    59→  return {
    60→    ca,
    61→    chains: [...chainSet],
    62→    numAtoms,
    63→    numResidues: residueSet.size,
    64→    title: title.replace(/\s+/g, " ").trim(),
    65→  };
    66→}

    68→/**
    69→ * Split a multi-model PDB file (NMR ensemble) into individual model texts.
    70→ * Returns an array of PDB text strings, one per MODEL record.
    71→ * If the file has no MODEL records, returns a single-element array with the
    72→ * original text.
    73→ */
    74→export function splitModels(pdb: string): { modelNum: number; pdbText: string }[] {
    75→  const lines = pdb.split(/\r?\n/);
    76→  const models: { modelNum: number; pdbText: string }[] = [];
    77→  let currentModel: number | null = null;
    78→  let currentLines: string[] = [];
    79→  let headerLines: string[] = [];
    80→
    81→  for (const line of lines) {
    82→    const rec = line.substring(0, 6).trim();
    83→    if (rec === "MODEL") {
    84→      currentModel = parseInt(line.substring(10, 14), 10) || models.length + 1;
    85→      currentLines = [...headerLines];
    86→    } else if (rec === "ENDMDL") {
    87→      if (currentModel !== null) {
    88→        models.push({ modelNum: currentModel, pdbText: currentLines.join("\n") });
    89→        currentModel = null;
    90→        currentLines = [];
    91→      }
    92→    } else if (currentModel !== null) {
    93→      currentLines.push(line);
    94→    } else {
    95→      // Collect header lines (before first MODEL) to prepend to each model.
    96→      if (rec !== "END" && rec !== "ENDMDL") {
    97→        headerLines.push(line);
    98→      }
    99→    }
   100→  }
   101→
   102→  // If no MODEL records found, return the original as a single model.
   103→  if (models.length === 0) {
   104→    return [{ modelNum: 1, pdbText: pdb }];
   105→  }
   106→
   107→  return models;
   108→}

   110→/**
   111→ * Match CA atoms of two structures by (chain, resSeq). Returns paired coordinates
   112→ * for the common subset. Falls back to sequential pairing (in chain order) when
   113→ * residue numbers do not overlap, which still gives a reasonable superposition
   114→ * for homologous structures.
   115→ */
   116→export function matchCAAtoms(
   117→  ref: CAAtom[],
   118→  mob: CAAtom[],
   119→  refChain?: string,
   120→  mobChain?: string
   121→): { refCoords: number[][]; mobCoords: number[][]; count: number } {
   122→  const refFiltered = refChain ? ref.filter((a) => a.chain === refChain) : ref;
   123→  const mobFiltered = mobChain ? mob.filter((a) => a.chain === mobChain) : mob;
   124→
   125→  const refMap = new Map<string, CAAtom>();
   126→  for (const a of refFiltered) refMap.set(`${a.chain}:${a.resSeq}`, a);
   127→
   128→  const refCoords: number[][] = [];
   129→  const mobCoords: number[][] = [];
   130→
   131→  for (const m of mobFiltered) {
   132→    const r = refMap.get(`${m.chain}:${m.resSeq}`);
   133→    if (r) {
   134→      refCoords.push([r.x, r.y, r.z]);
   135→      mobCoords.push([m.x, m.y, m.z]);
   136→    }
   137→  }
   138→
   139→  // Fallback: if no residue-number overlap, pair sequentially by order.
   140→  if (refCoords.length < 3 && refFiltered.length && mobFiltered.length) {
   141→    refCoords.length = 0;
   142→    mobCoords.length = 0;
   143→    const n = Math.min(refFiltered.length, mobFiltered.length);
   144→    for (let i = 0; i < n; i++) {
   145→      const r = refFiltered[i];
   146→      const m = mobFiltered[i];
   147→      refCoords.push([r.x, r.y, r.z]);
   148→      mobCoords.push([m.x, m.y, m.z]);
   149→    }
   150→  }
   151→
   152→  return { refCoords, mobCoords, count: refCoords.length };
   153→}

   180→/**
   181→ * Smith-Waterman local sequence alignment.
   182→ * Returns aligned pairs of indices (refIdx, mobIdx) for matched positions.
   183→ */
   184→function smithWaterman(
   185→  seq1: string,
   186→  seq2: string,
   187→  gapPenalty = -5
   188→): { pairs: [number, number][]; score: number } {
   189→  const m = seq1.length;
   190→  const n = seq2.length;
   191→  if (m === 0 || n === 0) return { pairs: [], score: 0 };
   192→
   193→  // DP matrix.
   194→  const H: number[][] = Array.from({ length: m + 1 }, () => new Array(n + 1).fill(0));
   195→  const trace: number[][] = Array.from({ length: m + 1 }, () => new Array(n + 1).fill(0));
   196→  // trace: 0=stop, 1=diag (match), 2=up (gap in seq2), 3=left (gap in seq1)
   197→
   198→  let maxScore = 0;
   199→  let maxI = 0;
   200→  let maxJ = 0;
   201→
   202→  for (let i = 1; i <= m; i++) {
   203→    const a = seq1[i - 1];
   204→    const row_a = BLOSUM62[a];
   205→    for (let j = 1; j <= n; j++) {
   206→      const b = seq2[j - 1];
   207→      const subScore = row_a ? (row_a[b] ?? -4) : -4;
   208→      const diag = H[i - 1][j - 1] + subScore;
   209→      const up = H[i - 1][j] + gapPenalty;
   210→      const left = H[i][j - 1] + gapPenalty;
   211→      const best = Math.max(0, diag, up, left);
   212→      H[i][j] = best;
   213→      if (best === 0) trace[i][j] = 0;
   214→      else if (best === diag) trace[i][j] = 1;
   215→      else if (best === up) trace[i][j] = 2;
   216→      else trace[i][j] = 3;
   217→
   218→      if (best > maxScore) {
   219→        maxScore = best;
   220→        maxI = i;
   221→        maxJ = j;
   222→      }
   223→    }
   224→  }
   225→
   226→  // Traceback from max.
   227→  const pairs: [number, number][] = [];
   228→  let i = maxI;
   229→  let j = maxJ;
   230→  while (i > 0 && j > 0 && H[i][j] > 0) {
   231→    if (trace[i][j] === 1) {
   232→      pairs.push([i - 1, j - 1]);
   233→      i--;
   234→      j--;
   235→    } else if (trace[i][j] === 2) {
   236→      i--;
   237→    } else {
   238→      j--;
   239→    }
   240→  }
   241→
   242→  pairs.reverse();
   243→  return { pairs, score: maxScore };
   244→}

   246→/**
   247→ * Needleman-Wunsch global sequence alignment.
   248→ * Aligns the full length of both sequences (unlike Smith-Waterman which finds
   249→ * the best local match). Better for comparing homologous structures where we
   250→ * want to find the correspondence between ALL residues, not just a high-scoring
   251→ * subregion. Avoids issues with internal sequence repeats.
   252→ */
   253→function needlemanWunsch(
   254→  seq1: string,
   255→  seq2: string,
   256→  gapPenalty = -5
   257→): { pairs: [number, number][]; score: number; identity: number } {
   258→  const m = seq1.length;
   259→  const n = seq2.length;
   260→  if (m === 0 || n === 0) return { pairs: [], score: 0, identity: 0 };
   261→
   262→  // DP matrix — no zeroing (global alignment, not local).
   263→  const H: number[][] = Array.from({ length: m + 1 }, () => new Array(n + 1).fill(0));
   264→  const trace: number[][] = Array.from({ length: m + 1 }, () => new Array(n + 1).fill(0));
   265→  // trace: 0=stop (only for [0,0]), 1=diag (match), 2=up (gap in seq2), 3=left (gap in seq1)
   266→
   267→  // Initialize first row and column with gap penalties.
   268→  for (let i = 0; i <= m; i++) H[i][0] = i * gapPenalty;
   269→  for (let j = 0; j <= n; j++) H[0][j] = j * gapPenalty;
   270→
   271→  for (let i = 1; i <= m; i++) {
   272→    const a = seq1[i - 1];
   273→    const row_a = BLOSUM62[a];
   274→    for (let j = 1; j <= n; j++) {
   275→      const b = seq2[j - 1];
   276→      const subScore = row_a ? (row_a[b] ?? -4) : -4;
   277→      const diag = H[i - 1][j - 1] + subScore;
   278→      const up = H[i - 1][j] + gapPenalty;
   279→      const left = H[i][j - 1] + gapPenalty;
   280→      // Global: take the best of the three (no zero option).
   281→      let best = diag;
   282→      let dir = 1;
   283→      if (up > best) { best = up; dir = 2; }
   284→      if (left > best) { best = left; dir = 3; }
   285→      H[i][j] = best;
   286→      trace[i][j] = dir;
   287→    }
   288→  }
   289→
   290→  // Traceback from [m, n] to [0, 0].
   291→  const pairs: [number, number][] = [];
   292→  let i = m;
   293→  let j = n;
   294→  let identical = 0;
   295→  while (i > 0 || j > 0) {
   296→    if (i > 0 && j > 0 && trace[i][j] === 1) {
   297→      pairs.push([i - 1, j - 1]);
   298→      if (seq1[i - 1] === seq2[j - 1]) identical++;
   299→      i--;
   300→      j--;
   301→    } else if (i > 0 && (j === 0 || trace[i][j] === 2)) {
   302→      i--; // gap in seq2
   303→    } else {
   304→      j--; // gap in seq1
   305→    }
   306→  }
   307→
   308→  pairs.reverse();
   309→  const identity = pairs.length > 0 ? identical / pairs.length : 0;
   310→  return { pairs, score: H[m][n], identity };
   311→}

   313→/**
   314→ * Match CA atoms by sequence alignment.
   315→ * Tries both Smith-Waterman (local) and Needleman-Wunsch (global) alignments,
   316→ * and picks the one with the better (lower) RMSD after Kabsch superposition.
   317→ * Used when residue numbers don't overlap or produce poor alignment.
   318→ */
   319→export function matchCABySequence(
   320→  ref: CAAtom[],
   321→  mob: CAAtom[],
   322→  refChain?: string,
   323→  mobChain?: string
   324→): { refCoords: number[][]; mobCoords: number[][]; count: number; alignScore: number } {
   325→  const refFiltered = refChain ? ref.filter((a) => a.chain === refChain) : ref;
   326→  const mobFiltered = mobChain ? mob.filter((a) => a.chain === mobChain) : mob;
   327→
   328→  if (refFiltered.length < 3 || mobFiltered.length < 3) {
   329→    return { refCoords: [], mobCoords: [], count: 0, alignScore: 0 };
   330→  }
   331→
   332→  // Build single-letter sequences from resName.
   333→  const toSeq = (atoms: CAAtom[]) =>
   334→    atoms.map((a) => AA3TO1[a.resName] || "X").join("");
   335→
   336→  const seq1 = toSeq(refFiltered);
   337→  const seq2 = toSeq(mobFiltered);
   338→
   339→  // Helper: convert pairs to coordinate arrays.
   340→  const pairsToCoords = (pairs: [number, number][]) => {
   341→    const rc: number[][] = [];
   342→    const mc: number[][] = [];
   343→    for (const [ri, mi] of pairs) {
   344→      const r = refFiltered[ri];
   345→      const m = mobFiltered[mi];
   346→      rc.push([r.x, r.y, r.z]);
   347→      mc.push([m.x, m.y, m.z]);
   348→    }
   349→    return { refCoords: rc, mobCoords: mc };
   350→  };
   351→
   352→  // Run both local (Smith-Waterman) and global (Needleman-Wunsch) alignments.
   353→  const swResult = smithWaterman(seq1, seq2);
   354→  const nwResult = needlemanWunsch(seq1, seq2);
   355→
   356→  const swCoords = pairsToCoords(swResult.pairs);
   357→  const nwCoords = pairsToCoords(nwResult.pairs);
   358→
   359→  // Pick the alignment with lower RMSD after Kabsch superposition.
   360→  // If both have enough pairs, compare RMSD; otherwise pick the one with more pairs.
   361→  let bestCoords = swCoords;
   362→  let bestScore = swResult.score;
   363→  let bestCount = swCoords.refCoords.length;
   364→
   365→  if (nwCoords.refCoords.length >= 3) {
   366→    const nwKabsch = kabsch(nwCoords.refCoords, nwCoords.mobCoords);
   367→    if (nwKabsch) {
   368→      if (bestCount >= 3) {
   369→        const swKabsch = kabsch(swCoords.refCoords, swCoords.mobCoords);
   370→        // Prefer the alignment with lower RMSD (if both have ≥3 pairs).
   371→        if (!swKabsch || nwKabsch.rmsd < swKabsch.rmsd) {
   372→          bestCoords = nwCoords;
   373→          bestScore = nwResult.score;
   374→          bestCount = nwCoords.refCoords.length;
   375→        }
   376→      } else {
   377→        bestCoords = nwCoords;
   378→        bestScore = nwResult.score;
   379→        bestCount = nwCoords.refCoords.length;
   380→      }
   381→    }
   382→  }
   383→
   384→  return {
   385→    refCoords: bestCoords.refCoords,
   386→    mobCoords: bestCoords.mobCoords,
   387→    count: bestCount,
   388→    alignScore: bestScore,
   389→  };
   390→}
   391→function centroid(points: number[][]): number[] {
   392→  const n = points.length;
   393→  if (n === 0) return [0, 0, 0];
   394→  let sx = 0,
   395→    sy = 0,
   396→    sz = 0;
   397→  for (const p of points) {
   398→    sx += p[0];
   399→    sy += p[1];
   400→    sz += p[2];
   401→  }
   402→  return [sx / n, sy / n, sz / n];
   403→}

   405→/**
   406→ * Multiply two matrices (m x n) and (n x p) -> (m x p).
   407→ */
   408→function matMul(a: number[][], b: number[][]): number[][] {
   409→  const m = a.length;
   410→  const n = b.length;
   411→  const p = b[0].length;
   412→  const out = Array.from({ length: m }, () => new Array(p).fill(0));
   413→  for (let i = 0; i < m; i++) {
   414→    for (let k = 0; k < n; k++) {
   415→      const aik = a[i][k];
   416→      if (aik === 0) continue;
   417→      for (let j = 0; j < p; j++) {
   418→        out[i][j] += aik * b[k][j];
   419→      }
   420→    }
   421→  }
   422→  return out;
   423→}
   424→
   425→function transpose(a: number[][]): number[][] {
   426→  const rows = a.length;
   427→  const cols = a[0].length;
   428→  const out = Array.from({ length: cols }, () => new Array(rows).fill(0));
   429→  for (let i = 0; i < rows; i++) for (let j = 0; j < cols; j++) out[j][i] = a[i][j];
   430→  return out;
   431→}

   433→/**
   434→ * SVD of a 3x3 matrix using the Jacobi eigenvalue iteration on A^T A.
   435→ * Returns { U: 3x3, S: number[3], V: 3x3 } such that A = U * diag(S) * V^T.
   436→ * This is robust enough for the 3x3 covariance matrices we encounter.
   437→ */
   438→function svd3(a: number[][]): { U: number[][]; S: number[]; V: number[][] } {
   439→  // Compute A^T A (3x3 symmetric) -> eigendecomposition gives V and singular values.
   440→  const At = transpose(a);
   441→  const AtA = matMul(At, a);
   442→  const { eigenvectors: V, eigenvalues } = jacobiEigen3(AtA);
   443→  // Singular values are sqrt of eigenvalues (clamp tiny negatives to 0).
   444→  const S = eigenvalues.map((v) => Math.sqrt(Math.max(0, v)));
   445→  // U columns: u_i = A v_i / s_i (for non-zero s_i)
   446→  // Here A is the 3x3 matrix `a`, v_i is the i-th eigenvector (column of V).
   447→  // We compute Av_i directly (matrix-vector product).
   448→  const U: number[][] = [
   449→    [0, 0, 0],
   450→    [0, 0, 0],
   451→    [0, 0, 0],
   452→  ];
   453→  for (let i = 0; i < 3; i++) {
   454→    // vi is the i-th column of V (an eigenvector of A^T A).
   455→    const v0 = V[0][i];
   456→    const v1 = V[1][i];
   457→    const v2 = V[2][i];
   458→    // Compute A * vi (3x3 matrix times 3x1 column vector).
   459→    const Av0 = a[0][0] * v0 + a[0][1] * v1 + a[0][2] * v2;
   460→    const Av1 = a[1][0] * v0 + a[1][1] * v1 + a[1][2] * v2;
   461→    const Av2 = a[2][0] * v0 + a[2][1] * v1 + a[2][2] * v2;
   462→    if (S[i] > 1e-10) {
   463→      U[0][i] = Av0 / S[i];
   464→      U[1][i] = Av1 / S[i];
   465→      U[2][i] = Av2 / S[i];
   466→    } else {
   467→      // Degenerate column; pick an arbitrary orthonormal vector later.
   468→      U[0][i] = 0;
   469→      U[1][i] = 0;
   470→      U[2][i] = 0;
   471→    }
   472→  }
   473→  // Ensure U is a proper rotation by orthonormalizing columns.
   474→  orthonormalize3(U);
   475→  return { U, S, V };
   476→}
   477→
   478→/** Orthonormalize the columns of a 3x3 matrix in place using Gram-Schmidt. */
   479→function orthonormalize3(m: number[][]) {
   480→  for (let i = 0; i < 3; i++) {
   481→    // v_i = v_i - sum_{j<i} (v_i . v_j) v_j
   482→    for (let j = 0; j < i; j++) {
   483→      let dot = 0;
   484→      for (let r = 0; r < 3; r++) dot += m[r][i] * m[r][j];
   485→      for (let r = 0; r < 3; r++) m[r][i] -= dot * m[r][j];
   486→    }
   487→    // normalize
   488→    let norm = 0;
   489→    for (let r = 0; r < 3; r++) norm += m[r][i] * m[r][i];
   490→    norm = Math.sqrt(norm) || 1;
   491→    for (let r = 0; r < 3; r++) m[r][i] /= norm;
   492→  }
   493→}

   495→/**
   496→ * Jacobi eigenvalue algorithm for a 3x3 symmetric matrix.
   497→ * Returns eigenvectors as columns and eigenvalues in ascending order.
   498→ */
   499→function jacobiEigen3(sym: number[][]): {
   500→  eigenvalues: number[];
   501→  eigenvectors: number[][];
   502→} {
   503→  // Copy into working arrays.
   504→  let a = [
   505→    [sym[0][0], sym[0][1], sym[0][2]],
   506→    [sym[1][0], sym[1][1], sym[1][2]],
   507→    [sym[2][0], sym[2][1], sym[2][2]],
   508→  ];
   509→  const v = [
   510→    [1, 0, 0],
   511→    [0, 1, 0],
   512→    [0, 0, 1],
   513→  ];
   514→
   515→  const maxIter = 60;
   516→  for (let iter = 0; iter < maxIter; iter++) {
   517→    // Find largest off-diagonal element.
   518→    let p = 0,
   519→      q = 1,
   520→      max = Math.abs(a[0][1]);
   521→    if (Math.abs(a[0][2]) > max) {
   522→      max = Math.abs(a[0][2]);
   523→      p = 0;
   524→      q = 2;
   525→    }
   526→    if (Math.abs(a[1][2]) > max) {
   527→      max = Math.abs(a[1][2]);
   528→      p = 1;
   529→      q = 2;
   530→    }
   531→    if (max < 1e-12) break;
   532→
   533→    const app = a[p][p];
   534→    const aqq = a[q][q];
   535→    const apq = a[p][q];
   536→    const phi = 0.5 * Math.atan2(2 * apq, aqq - app);
   537→    const c = Math.cos(phi);
   538→    const s = Math.sin(phi);
   539→
   540→    // Rotate.
   541→    for (let i = 0; i < 3; i++) {
   542→      const aiq = a[i][q];
   543→      const aip = a[i][p];
   544→      a[i][p] = c * aip - s * aiq;
   545→      a[i][q] = s * aip + c * aiq;
   546→    }
   547→    for (let j = 0; j < 3; j++) {
   548→      const apj = a[p][j];
   549→      const aqj = a[q][j];
   550→      a[p][j] = c * apj - s * aqj;
   551→      a[q][j] = s * apj + c * aqj;
   552→    }
   553→    for (let i = 0; i < 3; i++) {
   554→      const vip = v[i][p];
   555→      const viq = v[i][q];
   556→      v[i][p] = c * vip - s * viq;
   557→      v[i][q] = s * vip + c * viq;
   558→    }
   559→  }
   560→
   561→  const eigenvalues = [a[0][0], a[1][1], a[2][2]];
   562→  // eigenvectors are columns of v
   563→  const eigenvectors = [
   564→    [v[0][0], v[0][1], v[0][2]],
   565→    [v[1][0], v[1][1], v[1][2]],
   566→    [v[2][0], v[2][1], v[2][2]],
   567→  ];
   568→  return { eigenvalues, eigenvectors };
   569→}

   571→export interface KabschResult {
   572→  rotation: number[][]; // 3x3
   573→  translation: number[]; // 3 (pre-subtract mobile centroid, post-add ref centroid)
   574→  rmsd: number;
   575→  transform: number[][]; // 4x4 row-major homogeneous transform applied to mobile
   576→  count: number;
   577→  tmScore: number; // 0..1, >0.5 generally implies same fold
   578→}
   579→
   580→/**
   581→ * Kabsch algorithm: compute the optimal rigid-body transform that maps the
   582→ * mobile points onto the reference points (least-squares, RMSD-minimizing).
   583→ * Returns a 4x4 homogeneous transform (row-major) such that:
   584→ *   [x', y', z', 1] = transform @ [x, y, z, 1]
   585→ */
   586→export function kabsch(
   587→  refCoords: number[][],
   588→  mobCoords: number[][]
   589→): KabschResult | null {
   590→  const n = refCoords.length;
   591→  if (n < 3 || n !== mobCoords.length) return null;
   592→
   593→  const refC = centroid(refCoords);
   594→  const mobC = centroid(mobCoords);
   595→
   596→  // Center both point sets.
   597→  const refCentered = refCoords.map((p) => [p[0] - refC[0], p[1] - refC[1], p[2] - refC[2]]);
   598→  const mobCentered = mobCoords.map((p) => [p[0] - mobC[0], p[1] - mobC[1], p[2] - mobC[2]]);
   599→
   600→  // Covariance H = sum mob_i (ref_i)^T  -> 3x3
   601→  // We want R minimizing sum || R*mob - ref ||^2. Standard Kabsch uses H = P^T Q
   602→  // where P = mob, Q = ref. R = V * (det sign) * U^T from SVD of H.
   603→  const H = [
   604→    [0, 0, 0],
   605→    [0, 0, 0],
   606→    [0, 0, 0],
   607→  ];
   608→  for (let i = 0; i < n; i++) {
   609→    const m = mobCentered[i];
   610→    const r = refCentered[i];
   611→    H[0][0] += m[0] * r[0];
   612→    H[0][1] += m[0] * r[1];
   613→    H[0][2] += m[0] * r[2];
   614→    H[1][0] += m[1] * r[0];
   615→    H[1][1] += m[1] * r[1];
   616→    H[1][2] += m[1] * r[2];
   617→    H[2][0] += m[2] * r[0];
   618→    H[2][1] += m[2] * r[1];
   619→    H[2][2] += m[2] * r[2];
   620→  }
   621→
   622→  const { U, S, V } = svd3(H);
   623→
   624→  // Correct for reflection: d = sign(det(V * U^T))
   625→  const VUt = matMul(V, transpose(U));
   626→  let d = VUt[0][0] * (VUt[1][1] * VUt[2][2] - VUt[1][2] * VUt[2][1]) -
   627→    VUt[0][1] * (VUt[1][0] * VUt[2][2] - VUt[1][2] * VUt[2][0]) +
   628→    VUt[0][2] * (VUt[1][0] * VUt[2][1] - VUt[1][1] * VUt[2][0]);
   629→  d = d > 0 ? 1 : d < 0 ? -1 : 0;
   630→
   631→  // Build the reflection correction diagonal.
   632→  const D = [
   633→    [1, 0, 0],
   634→    [0, 1, 0],
   635→    [0, 0, d],
   636→  ];
   637→
   638→  // Rotation: R = V * D * U^T
   639→  const R = matMul(matMul(V, D), transpose(U));
   640→
   641→  // Build 4x4 homogeneous transform: x' = R (x - mobC) + refC
   642→  // i.e. x' = R x + (refC - R mobC)
   643→  const t: number[] = [
   644→    refC[0] - (R[0][0] * mobC[0] + R[0][1] * mobC[1] + R[0][2] * mobC[2]),
   645→    refC[1] - (R[1][0] * mobC[0] + R[1][1] * mobC[1] + R[1][2] * mobC[2]),
   646→    refC[2] - (R[2][0] * mobC[0] + R[2][1] * mobC[1] + R[2][2] * mobC[2]),
   647→  ];
   648→
   649→  const transform: number[][] = [
   650→    [R[0][0], R[0][1], R[0][2], t[0]],
   651→    [R[1][0], R[1][1], R[1][2], t[1]],
   652→    [R[2][0], R[2][1], R[2][2], t[2]],
   653→    [0, 0, 0, 1],
   654→  ];
   655→
   656→  // RMSD over the aligned subset.
   657→  let sumSq = 0;
   658→  for (let i = 0; i < n; i++) {
   659→    const m = mobCoords[i];
   660→    const mx = R[0][0] * m[0] + R[0][1] * m[1] + R[0][2] * m[2] + t[0];
   661→    const my = R[1][0] * m[0] + R[1][1] * m[1] + R[1][2] * m[2] + t[1];
   662→    const mz = R[2][0] * m[0] + R[2][1] * m[1] + R[2][2] * m[2] + t[2];
   663→    const dx = mx - refCoords[i][0];
   664→    const dy = my - refCoords[i][1];
   665→    const dz = mz - refCoords[i][2];
   666→    sumSq += dx * dx + dy * dy + dz * dz;
   667→  }
   668→  const rmsd = Math.sqrt(sumSq / n);
   669→
   670→  // TM-score: a length-normalized similarity metric that is more robust than
   671→  // RMSD for distant homologs. d0 = 1.24*(L-15)^(1/3) - 1.8, clamped to >= 0.5.
   672→  // TM = (1/L) * sum_i [ 1 / (1 + (d_i/d0)^2) ]
   673→  // where L = length of the reference (target) structure and the sum is over
   674→  // all aligned pairs. TM ranges 0..1; >0.5 generally means same fold.
   675→  const Lnorm = n;
   676→  const d0 = Math.max(0.5, 1.24 * Math.pow(Math.max(1, Lnorm - 15), 1 / 3) - 1.8);
   677→  let tmSum = 0;
   678→  for (let i = 0; i < n; i++) {
   679→    const m = mobCoords[i];
   680→    const mx = R[0][0] * m[0] + R[0][1] * m[1] + R[0][2] * m[2] + t[0];
   681→    const my = R[1][0] * m[0] + R[1][1] * m[1] + R[1][2] * m[2] + t[1];
   682→    const mz = R[2][0] * m[0] + R[2][1] * m[1] + R[2][2] * m[2] + t[2];
   683→    const dx = mx - refCoords[i][0];
   684→    const dy = my - refCoords[i][1];
   685→    const dz = mz - refCoords[i][2];
   686→    const d2 = dx * dx + dy * dy + dz * dz;
   687→    tmSum += 1 / (1 + d2 / (d0 * d0));
   688→  }
   689→  const tmScore = tmSum / Lnorm;
   690→
   691→  return { rotation: R, translation: t, rmsd, transform, count: n, tmScore };
   692→}
   693→
   694→/**
   695→ * Apply a 4x4 homogeneous transform (row-major) to a parsed PDB text and return
   696→ * a new PDB text with transformed ATOM/HETATM coordinate columns. Only the
   697→ * coordinates are rewritten; everything else is preserved.
   698→ */
   699→export function applyTransformToPdb(pdb: string, transform: number[][]): string {
   700→  const R = [
   701→    [transform[0][0], transform[0][1], transform[0][2]],
   702→    [transform[1][0], transform[1][1], transform[1][2]],
   703→    [transform[2][0], transform[2][1], transform[2][2]],
   704→  ];
   705→  const t = [transform[0][3], transform[1][3], transform[2][3]];
   706→
   707→  const out: string[] = [];
   708→  const lines = pdb.split(/\r?\n/);
   709→  for (const line of lines) {
   710→    const rec = line.substring(0, 6).trim();
   711→    if (rec === "ATOM" || rec === "HETATM") {
   712→      const x = parseFloat(line.substring(30, 38));
   713→      const y = parseFloat(line.substring(38, 46));
   714→      const z = parseFloat(line.substring(46, 54));
   715→      if (Number.isNaN(x) || Number.isNaN(y) || Number.isNaN(z)) {
   716→        out.push(line);
   717→        continue;
   718→      }
   719→      const nx = R[0][0] * x + R[0][1] * y + R[0][2] * z + t[0];
   720→      const ny = R[1][0] * x + R[1][1] * y + R[1][2] * z + t[1];
   721→      const nz = R[2][0] * x + R[2][1] * y + R[2][2] * z + t[2];
   722→      // Reformat coordinate columns (8.3f) to preserve PDB column alignment.
   723→      const xs = nx.toFixed(3).padStart(8, " ");
   724→      const ys = ny.toFixed(3).padStart(8, " ");
   725→      const zs = nz.toFixed(3).padStart(8, " ");
   726→      out.push(line.substring(0, 30) + xs + ys + zs + line.substring(54));
   727→    } else {
   728→      out.push(line);
   729→    }
   730→  }
   731→  return out.join("\n");
   732→}

   747→export function compositionSummary(pdb: string, parsed: ParsedPdb): CompositionSummary {
   748→  const lines = pdb.split(/\r?\n/);
   749→  let numWaters = 0;
   750→  let helixCount = 0;
   751→  let sheetCount = 0;
   752→  const resMap = new Map<string, number>();
   753→  const chainSet = new Set<string>();
   754→
   755→  for (const line of lines) {
   756→    const rec = line.substring(0, 6).trim();
   757→    if (rec === "ATOM" || rec === "HETATM") {
   758→      const resName = line.substring(17, 20).trim();
   759→      const chain = line.substring(21, 22).trim() || " ";
   760→      chainSet.add(chain);
   761→      if (rec === "HETATM" && resName === "HOH") numWaters++;
   762→      else if (rec === "ATOM") {
   763→        resMap.set(resName, (resMap.get(resName) || 0) + 1);
   764→      }
   765→    } else if (rec === "HELIX") {
   766→      helixCount++;
   767→    } else if (rec === "SHEET") {
   768→      sheetCount++;
   769→    }
   770→  }
   771→
   772→  const residueCounts = [...resMap.entries()]
   773→    .map(([resName, count]) => ({ resName, count }))
   774→    .sort((a, b) => b.count - a.count);
   775→
   776→  return {
   777→    chains: [...chainSet],
   778→    numAtoms: parsed.numAtoms,
   779→    numResidues: parsed.numResidues,
   780→    numWaters,
   781→    residueCounts: residueCounts.slice(0, 20),
   782→    helixCount,
   783→    sheetCount,
   784→  };
   785→}

   802→/**
   803→ * Detect HETATM ligand records (excluding waters). Returns one entry per
   804→ * unique (resName, chain, resSeq) ligand, with its centroid for "focus" actions.
   805→ */
   806→export function detectLigands(pdb: string): LigandInfo[] {
   807→  const lines = pdb.split(/\r?\n/);
   808→  const map = new Map<string, LigandInfo>();
   809→
   810→  for (const line of lines) {
   811→    const rec = line.substring(0, 6).trim();
   812→    if (rec !== "HETATM") continue;
   813→    const resName = line.substring(17, 20).trim();
   814→    if (!resName || resName === "HOH") continue;
   815→    const chain = line.substring(21, 22).trim() || " ";
   816→    const resSeq = parseInt(line.substring(22, 26), 10);
   817→    if (Number.isNaN(resSeq)) continue;
   818→    const x = parseFloat(line.substring(30, 38));
   819→    const y = parseFloat(line.substring(38, 46));
   820→    const z = parseFloat(line.substring(46, 54));
   821→    if (Number.isNaN(x) || Number.isNaN(y) || Number.isNaN(z)) continue;
   822→
   823→    const key = `${resName}:${chain}:${resSeq}`;
   824→    const existing = map.get(key);
   825→    if (existing) {
   826→      existing.numAtoms++;
   827→      existing.centerX += x;
   828→      existing.centerY += y;
   829→      existing.centerZ += z;
   830→    } else {
   831→      map.set(key, {
   832→        resName,
   833→        chain,
   834→        resSeq,
   835→        numAtoms: 1,
   836→        centerX: x,
   837→        centerY: y,
   838→        centerZ: z,
   839→      });
   840→    }
   841→  }
   842→
   843→  // Finalize centroids.
   844→  for (const lig of map.values()) {
   845→    if (lig.numAtoms > 0) {
   846→      lig.centerX /= lig.numAtoms;
   847→      lig.centerY /= lig.numAtoms;
   848→      lig.centerZ /= lig.numAtoms;
   849→    }
   850→  }
   851→
   852→  return [...map.values()].sort((a, b) => b.numAtoms - a.numAtoms);
   853→}

   855→/**
   856→ * Find the centroid of all atoms in the structure (for "reset view" focus).
   857→ */
   858→export function structureCentroid(pdb: string): { x: number; y: number; z: number } | null {
   859→  const parsed = parsePdb(pdb);
   860→  if (parsed.ca.length === 0) return null;
   861→  let sx = 0, sy = 0, sz = 0;
   862→  for (const a of parsed.ca) {
   863→    sx += a.x;
   864→    sy += a.y;
   865→    sz += a.z;
   866→  }
   867→  const n = parsed.ca.length;
   868→  return { x: sx / n, y: sy / n, z: sz / n };
   869→}

   871→export interface HBond {
   872→  donorChain: string;

   888→
   889→// Common hydrogen-bond donor atoms (backbone N + sidechain donors).
   890→const HBOND_DONORS = new Set(["N", "ND1", "ND2", "NE", "NE1", "NE2", "NH1", "NH2", "NZ", "OG", "OG1", "OH", "SG"]);
   891→// Common hydrogen-bond acceptor atoms (backbone O + sidechain acceptors).
   892→const HBOND_ACCEPTORS = new Set(["O", "OD1", "OD2", "OE1", "OE2", "OG", "OG1", "OH", "SD", "ND1", "NE2"]);
   893→
   894→/**
   895→ * Detect hydrogen bonds in a PDB structure using a simple geometric criterion:
   896→ * donor-acceptor distance ≤ 3.5 Å and donor not in the same residue as acceptor.
   897→ * This is an approximation (no angle check) suitable for visualization.
   898→ */
   899→export function detectHBonds(pdb: string, maxDist = 3.5): HBond[] {
   900→  const lines = pdb.split(/\r?\n/);
   901→  const donors: any[] = [];
   902→  const acceptors: any[] = [];
   903→
   904→  for (const line of lines) {
   905→    const rec = line.substring(0, 6).trim();
   906→    if (rec !== "ATOM" && rec !== "HETATM") continue;
   907→    const atomName = line.substring(12, 16).trim();
   908→    const resName = line.substring(17, 20).trim();
   909→    const chain = line.substring(21, 22).trim() || " ";
   910→    const resSeq = parseInt(line.substring(22, 26), 10);
   911→    if (Number.isNaN(resSeq)) continue;
   912→    const x = parseFloat(line.substring(30, 38));
   913→    const y = parseFloat(line.substring(38, 46));
   914→    const z = parseFloat(line.substring(46, 54));
   915→    if (Number.isNaN(x) || Number.isNaN(y) || Number.isNaN(z)) continue;
   916→
   917→    const atom = { atomName, resName, chain, resSeq, x, y, z };
   918→    if (HBOND_DONORS.has(atomName)) donors.push(atom);
   919→    if (HBOND_ACCEPTORS.has(atomName)) acceptors.push(atom);
   920→  }
   921→
   922→  const bonds: HBond[] = [];
   923→  const seen = new Set<string>();
   924→
   925→  for (const d of donors) {
   926→    for (const a of acceptors) {
   927→      // Skip same-residue.
   928→      if (d.chain === a.chain && d.resSeq === a.resSeq) continue;
   929→      // Skip backbone N-O in adjacent residues (those are still valid HBonds, keep them).
   930→      const dx = d.x - a.x;
   931→      const dy = d.y - a.y;
   932→      const dz = d.z - a.z;
   933→      const dist = Math.sqrt(dx * dx + dy * dy + dz * dz);
   934→      if (dist > maxDist || dist < 1.0) continue;
   935→
   936→      // Deduplicate by unordered pair key.
   937→      const key = [`${d.chain}:${d.resSeq}:${d.atomName}`, `${a.chain}:${a.resSeq}:${a.atomName}`]
   938→        .sort()
   939→        .join("|");
   940→      if (seen.has(key)) continue;
   941→      seen.add(key);
   942→
   943→      bonds.push({
   944→        donorChain: d.chain,
   945→        donorResSeq: d.resSeq,
   946→        donorResName: d.resName,
   947→        donorAtom: d.atomName,
   948→        donorX: d.x,
   949→        donorY: d.y,
   950→        donorZ: d.z,
   951→        acceptorChain: a.chain,
   952→        acceptorResSeq: a.resSeq,
   953→        acceptorResName: a.resName,
   954→        acceptorAtom: a.atomName,
   955→        acceptorX: a.x,
   956→        acceptorY: a.y,
   957→        acceptorZ: a.z,
   958→        distance: dist,
   959→      });
   960→    }
   961→  }
   962→
   963→  return bonds.sort((a, b) => a.distance - b.distance);
   964→}

  1024→export interface RamachandranPoint {
  1025→  chain: string;
  1026→  resSeq: number;
  1027→  resName: string;
  1028→  phi: number | null; // degrees
  1029→  psi: number | null; // degrees
  1030→  region: "core" | "allowed" | "generous" | "disallowed";
  1031→}
  1032→
  1033→/**
  1034→ * Compute phi/psi backbone dihedral angles for each residue.
  1035→ * phi(i) = dihedral(C(i-1), N(i), CA(i), C(i))
  1036→ * psi(i) = dihedral(N(i), CA(i), C(i), N(i+1))
  1037→ */
  1038→export function computeRamachandran(pdb: string): RamachandranPoint[] {
  1039→  const lines = pdb.split(/\r?\n/);
  1040→  const residueMap = new Map<string, BackboneAtom>();
  1041→
  1042→  for (const line of lines) {
  1043→    const rec = line.substring(0, 6).trim();
  1044→    if (rec !== "ATOM") continue;
  1045→    const atomName = line.substring(12, 16).trim();
  1046→    const resName = line.substring(17, 20).trim();
  1047→    const chain = line.substring(21, 22).trim() || " ";
  1048→    const resSeq = parseInt(line.substring(22, 26), 10);
  1049→    if (Number.isNaN(resSeq)) continue;
  1050→    if (atomName !== "N" && atomName !== "CA" && atomName !== "C") continue;
  1051→    const x = parseFloat(line.substring(30, 38));
  1052→    const y = parseFloat(line.substring(38, 46));
  1053→    const z = parseFloat(line.substring(46, 54));
  1054→    if (Number.isNaN(x) || Number.isNaN(y) || Number.isNaN(z)) continue;
  1055→
  1056→    const key = `${chain}:${resSeq}`;
  1057→    if (!residueMap.has(key)) {
  1058→      residueMap.set(key, { chain, resSeq, resName });
  1059→    }
  1060→    const res = residueMap.get(key)!;
  1061→    if (atomName === "N") res.n = [x, y, z];
  1062→    else if (atomName === "CA") res.ca = [x, y, z];
  1063→    else if (atomName === "C") res.c = [x, y, z];
  1064→  }
  1065→
  1066→  // Sort residues by chain then resSeq.
  1067→  const residues = [...residueMap.values()].sort((a, b) => {
  1068→    if (a.chain !== b.chain) return a.chain.localeCompare(b.chain);
  1069→    return a.resSeq - b.resSeq;
  1070→  });
  1071→
  1072→  const points: RamachandranPoint[] = [];
  1073→
  1074→  for (let i = 0; i < residues.length; i++) {
  1075→    const cur = residues[i];
  1076→    const prev = i > 0 ? residues[i - 1] : null;
  1077→    const next = i < residues.length - 1 ? residues[i + 1] : null;
  1078→
  1079→    // phi requires prev C, cur N, cur CA, cur C
  1080→    let phi: number | null = null;
  1081→    if (prev?.c && cur.n && cur.ca && cur.c) {
  1082→      // Only valid if prev and cur are on the same chain (peptide bond).
  1083→      if (prev.chain === cur.chain) {
  1084→        phi = dihedral(prev.c, cur.n, cur.ca, cur.c);
  1085→      }
  1086→    }
  1087→
  1088→    // psi requires cur N, cur CA, cur C, next N
  1089→    let psi: number | null = null;
  1090→    if (cur.n && cur.ca && cur.c && next?.n) {
  1091→      if (cur.chain === next.chain) {
  1092→        psi = dihedral(cur.n, cur.ca, cur.c, next.n);
  1093→      }
  1094→    }
  1095→
  1096→    if (phi !== null || psi !== null) {
  1097→      points.push({
  1098→        chain: cur.chain,
  1099→        resSeq: cur.resSeq,
  1100→        resName: cur.resName,
  1101→        phi,
  1102→        psi,
  1103→        region: classifyRamachandran(phi, psi, cur.resName),
  1104→      });
  1105→    }
  1106→  }
  1107→
  1108→  return points;
  1109→}
  1110→
  1111→/**
  1112→ * Classify a (phi, psi) point into Ramachandran regions.
  1113→ * Simplified: checks if the point is in the favored alpha-helix / beta-sheet
  1114→ * basins. Glycine is treated more permissively.
  1115→ */
  1116→function classifyRamachandran(
  1117→  phi: number | null,
  1118→  psi: number | null,
  1119→  resName: string
  1120→): "core" | "allowed" | "generous" | "disallowed" {
  1121→  if (phi === null || psi === null) return "disallowed";
  1122→  // Normalize to -180..180.
  1123→  const p = ((phi + 180) % 360 + 360) % 360 - 180;
  1124→  const s = ((psi + 180) % 360 + 360) % 360 - 180;
  1125→
  1126→  const dist2 = (x: number, y: number) => (p - x) ** 2 + (s - y) ** 2;
  1127→
  1128→  // Alpha helix basin: phi≈-57, psi≈-47
  1129→  const dHelix = Math.sqrt(dist2(-57, -47));
  1130→  // Beta sheet basin: phi≈-119, psi≈113  (and phi≈-139, psi≈135)
  1131→  const dBeta = Math.sqrt(Math.min(dist2(-119, 113), dist2(-139, 135)));
  1132→  // Left-handed helix: phi≈57, psi≈47 (rare, mostly Gly)
  1133→  const dLHelix = Math.sqrt(dist2(57, 47));
  1134→
  1135→  const isGly = resName === "GLY";
  1136→  const minDist = Math.min(dHelix, dBeta, dLHelix);
  1137→
  1138→  if (minDist < 25) return "core";
  1139→  if (minDist < 50) return "allowed";
  1140→  if (minDist < 80 || (isGly && dLHelix < 100)) return "generous";
  1141→  return "disallowed";
  1142→}

  1144→/**
  1145→ * Compute the dihedral angle between 4 points (a-b-c-d) in degrees.
  1146→ */
  1147→function dihedral(
  1148→  a: [number, number, number],
  1149→  b: [number, number, number],
  1150→  c: [number, number, number],
  1151→  d: [number, number, number]
  1152→): number {
  1153→  const sub = (x: number[], y: number[]) => [x[0] - y[0], x[1] - y[1], x[2] - y[2]];
  1154→  const cross = (x: number[], y: number[]) => [
  1155→    x[1] * y[2] - x[2] * y[1],
  1156→    x[2] * y[0] - x[0] * y[2],
  1157→    x[0] * y[1] - x[1] * y[0],
  1158→  ];
  1159→  const dot = (x: number[], y: number[]) => x[0] * y[0] + x[1] * y[1] + x[2] * y[2];
  1160→  const norm = (x: number[]) => Math.sqrt(dot(x, x));
  1161→
  1162→  const b1 = sub(b, a);
  1163→  const b2 = sub(c, b);
  1164→  const b3 = sub(d, c);
  1165→
  1166→  const m1 = cross(b1, b2);
  1167→  const m2 = cross(b2, b3);
  1168→  const x = dot(m1, m2);
  1169→  const y = dot(cross(m1, b2), m2) * (1 / (norm(b2) || 1));
  1170→
  1171→  return (Math.atan2(y, x) * 180) / Math.PI;
  1172→}

  1231→export function extractSequences(pdb: string): SequenceInfo[] {
  1232→  const lines = pdb.split(/\r?\n/);

  1234→
  1235→  // First try SEQRES.
  1236→  for (const line of lines) {
  1237→    if (line.substring(0, 6).trim() !== "SEQRES") continue;

  1241→    for (const r of residues) {

  1243→      seqMap.get(chain)!.push(one);
  1244→    }
  1245→  }
  1246→
  1247→  // If no SEQRES, fall back to ATOM records (first occurrence per residue).
  1248→  if (seqMap.size === 0) {
  1249→    const seen = new Map<string, string>();
  1250→    for (const line of lines) {
  1251→      if (line.substring(0, 6).trim() !== "ATOM") continue;
  1252→      const atomName = line.substring(12, 16).trim();
  1253→      if (atomName !== "CA") continue;
  1254→      const resName = line.substring(17, 20).trim();
  1255→      const chain = line.substring(21, 22).trim() || "A";

  1259→      if (seen.has(key)) continue;

  1264→    for (const [key, aa] of seen.entries()) {

  1270→      residues.sort((a, b) => a.resSeq - b.resSeq);

  1354→/**
  1355→ * Approximate per-residue SASA using a simple shrake-rupley-like approach.
  1356→ * For each residue's CA atom, count how many test points on a sphere around it
  1357→ * are not occluded by other atoms. This is a fast approximation suitable for
  1358→ * visualization (not publication-grade).
  1359→ *
  1360→ * @param pdb PDB text
  1361→ * @param probeRadius Water probe radius (default 1.4 Å)
  1362→ * @param numPoints Number of test points per residue (default 50, more = slower but accurate)
  1363→ */
  1364→export function computeSASA(
  1365→  pdb: string,
  1366→  probeRadius = 1.4,
  1367→  numPoints = 50
  1368→): ResidueSASA[] {
  1369→  const atoms = extractAllAtoms(pdb);
  1370→  const caAtoms = atoms.filter((a) => a.atomName === "CA" && a.isBackbone);
  1371→
  1372→  if (caAtoms.length === 0) return [];
  1373→
  1374→  // Precompute neighbor list: for each CA, find atoms within a cutoff.
  1375→  const cutoff = 15; // Å — only check atoms within this distance
  1376→  const results: ResidueSASA[] = [];
  1377→  let maxSasa = 0;
  1378→
  1379→  // Generate Fibonacci sphere points for even distribution.
  1380→  const points: [number, number, number][] = [];
  1381→  const golden = (1 + Math.sqrt(5)) / 2;
  1382→  for (let i = 0; i < numPoints; i++) {
  1383→    const t = i / numPoints;
  1384→    const phi = Math.acos(1 - 2 * t);
  1385→    const theta = 2 * Math.PI * i / golden;
  1386→    points.push([
  1387→      Math.sin(phi) * Math.cos(theta),
  1388→      Math.sin(phi) * Math.sin(theta),
  1389→      Math.cos(phi),
  1390→    ]);
  1391→  }
  1392→
  1393→  for (const ca of caAtoms) {
  1394→    // Test sphere radius = CA VDW + probe
  1395→    const testRadius = ca.vdwRadius + probeRadius + 3; // extend to sample neighborhood
  1396→    let accessible = 0;
  1397→
  1398→    for (const [px, py, pz] of points) {
  1399→      const tx = ca.x + px * testRadius;
  1400→      const ty = ca.y + py * testRadius;
  1401→      const tz = ca.z + pz * testRadius;
  1402→
  1403→      // Check if this test point is occluded by any other atom.
  1404→      let occluded = false;
  1405→      for (const other of atoms) {
  1406→        if (other.serial === ca.serial) continue;
  1407→        const dx = tx - other.x;
  1408→        const dy = ty - other.y;
  1409→        const dz = tz - other.z;
  1410→        const distSq = dx * dx + dy * dy + dz * dz;
  1411→        const occRadius = other.vdwRadius + probeRadius;
  1412→        if (distSq < occRadius * occRadius) {
  1413→          occluded = true;
  1414→          break;
  1415→        }
  1416→        // Early exit if too far (optimization).
  1417→        if (distSq > cutoff * cutoff) continue;
  1418→      }
  1419→
  1420→      if (!occluded) accessible++;
  1421→    }
  1422→
  1423→    // SASA proportional to fraction of accessible points × sphere area.
  1424→    const fraction = accessible / numPoints;
  1425→    const sphereArea = 4 * Math.PI * testRadius * testRadius;
  1426→    const sasa = fraction * sphereArea;
  1427→
  1428→    if (sasa > maxSasa) maxSasa = sasa;
  1429→
  1430→    results.push({
  1431→      chain: ca.chain,
  1432→      resSeq: ca.resSeq,
  1433→      resName: ca.resName,
  1434→      sasa,
  1435→      normalizedSasa: 0, // filled in after max known
  1436→      exposure: "intermediate",
  1437→    });
  1438→  }
  1439→
  1440→  // Normalize + classify.
  1441→  for (const r of results) {
  1442→    r.normalizedSasa = maxSasa > 0 ? r.sasa / maxSasa : 0;
  1443→    if (r.normalizedSasa < 0.25) r.exposure = "buried";
  1444→    else if (r.normalizedSasa > 0.6) r.exposure = "exposed";
  1445→    else r.exposure = "intermediate";
  1446→  }
  1447→
  1448→  return results;
  1449→}

  1451→export interface ClashInfo {
  1452→  atom1: AtomInfo;
  1453→  atom2: AtomInfo;
  1454→  distance: number;
  1455→  overlap: number; // Å of overlap (sum of VDW radii - distance)
  1456→  severity: "minor" | "moderate" | "severe";
  1457→}
  1458→
  1459→/**
  1460→ * Detect steric clashes: atom pairs where distance < sum of VDW radii - tolerance.
  1461→ */
  1462→export function detectClashes(pdb: string, tolerance = 0.4): ClashInfo[] {
  1463→  const atoms = extractAllAtoms(pdb);
  1464→  const clashes: ClashInfo[] = [];
  1465→  const cutoff = 4; // Å — only check pairs closer than this
  1466→
  1467→  // O(n²) but with cutoff optimization. For large structures, consider spatial hashing.
  1468→  for (let i = 0; i < atoms.length; i++) {
  1469→    for (let j = i + 1; j < atoms.length; j++) {
  1470→      const a = atoms[i];
  1471→      const b = atoms[j];
  1472→      // Skip atoms in the same residue (bonded).
  1473→      if (a.chain === b.chain && a.resSeq === b.resSeq) continue;
  1474→
  1475→      const dx = a.x - b.x;
  1476→      const dy = a.y - b.y;
  1477→      const dz = a.z - b.z;
  1478→      const distSq = dx * dx + dy * dy + dz * dz;
  1479→      if (distSq > cutoff * cutoff) continue;
  1480→
  1481→      const dist = Math.sqrt(distSq);
  1482→      const vdwSum = a.vdwRadius + b.vdwRadius;
  1483→      const overlap = vdwSum - dist - tolerance;
  1484→
  1485→      if (overlap > 0) {
  1486→        clashes.push({
  1487→          atom1: a,
  1488→          atom2: b,
  1489→          distance: dist,
  1490→          overlap,
  1491→          severity: overlap > 1.0 ? "severe" : overlap > 0.5 ? "moderate" : "minor",
  1492→        });
  1493→      }
  1494→    }
  1495→  }
  1496→
  1497→  return clashes.sort((a, b) => b.overlap - a.overlap);
  1498→}

  1500→export interface BondStats {
  1501→  count: number;
  1502→  meanLength: number;
  1503→  minLength: number;
  1504→  maxLength: number;
  1505→  outliers: { atom1: AtomInfo; atom2: AtomInfo; length: number; expected: number }[];
  1506→}

  1508→/**
  1509→ * Compute backbone bond length statistics (N-CA, CA-C, C-N).
  1510→ * Identifies bonds that deviate significantly from expected values.
  1511→ */
  1512→export function computeBondStats(pdb: string): {
  1513→  nCa: BondStats;
  1514→  caC: BondStats;
  1515→  cN: BondStats;
  1516→} {
  1517→  const atoms = extractAllAtoms(pdb);
  1518→  // Group by residue.
  1519→  const residueMap = new Map<string, { n?: AtomInfo; ca?: AtomInfo; c?: AtomInfo }>();
  1520→  for (const a of atoms) {
  1521→    if (!["N", "CA", "C"].includes(a.atomName)) continue;
  1522→    const key = `${a.chain}:${a.resSeq}`;
  1523→    if (!residueMap.has(key)) residueMap.set(key, {});
  1524→    const res = residueMap.get(key)!;
  1525→    if (a.atomName === "N") res.n = a;
  1526→    else if (a.atomName === "CA") res.ca = a;
  1527→    else if (a.atomName === "C") res.c = a;
  1528→  }
  1529→
  1530→  const residues = [...residueMap.values()];
  1531→
  1532→  const computeStats = (
  1533→    pairs: { a: AtomInfo; b: AtomInfo }[],
  1534→    expected: number
  1535→  ): BondStats => {
  1536→    if (pairs.length === 0) {
  1537→      return { count: 0, meanLength: 0, minLength: 0, maxLength: 0, outliers: [] };
  1538→    }
  1539→    const lengths = pairs.map((p) => {
  1540→      const dx = p.a.x - p.b.x;
  1541→      const dy = p.a.y - p.b.y;
  1542→      const dz = p.a.z - p.b.z;
  1543→      return Math.sqrt(dx * dx + dy * dy + dz * dz);
  1544→    });
  1545→    const mean = lengths.reduce((s, l) => s + l, 0) / lengths.length;
  1546→    const min = Math.min(...lengths);
  1547→    const max = Math.max(...lengths);
  1548→    const outliers = pairs
  1549→      .map((p, i) => ({ atom1: p.a, atom2: p.b, length: lengths[i], expected }))
  1550→      .filter((o) => Math.abs(o.length - expected) > 0.3); // >0.3 Å deviation
  1551→    return { count: lengths.length, meanLength: mean, minLength: min, maxLength: max, outliers };
  1552→  };
  1553→
  1554→  // N-CA and CA-C within each residue.
  1555→  const nCaPairs: { a: AtomInfo; b: AtomInfo }[] = [];
  1556→  const caCPairs: { a: AtomInfo; b: AtomInfo }[] = [];
  1557→  for (const res of residues) {
  1558→    if (res.n && res.ca) nCaPairs.push({ a: res.n, b: res.ca });
  1559→    if (res.ca && res.c) caCPairs.push({ a: res.ca, b: res.c });
  1560→  }
  1561→
  1562→  // C-N between consecutive residues (peptide bond).
  1563→  const cNPairs: { a: AtomInfo; b: AtomInfo }[] = [];
  1564→  // Iterate atoms in order and find C-N pairs across residues.
  1565→  const atomByChainRes = new Map<string, { n?: AtomInfo; c?: AtomInfo }>();
  1566→  for (const a of atoms) {
  1567→    if (a.atomName !== "N" && a.atomName !== "C") continue;
  1568→    const key = `${a.chain}:${a.resSeq}`;
  1569→    if (!atomByChainRes.has(key)) atomByChainRes.set(key, {});
  1570→    const entry = atomByChainRes.get(key)!;
  1571→    if (a.atomName === "N") entry.n = a;
  1572→    else entry.c = a;
  1573→  }
  1574→  // Sort by resSeq per chain.
  1575→  const chainResidues = new Map<string, { resSeq: number; n?: AtomInfo; c?: AtomInfo }[]>();
  1576→  for (const [key, entry] of atomByChainRes.entries()) {
  1577→    const [chain, resSeqStr] = key.split(":");
  1578→    if (!chainResidues.has(chain)) chainResidues.set(chain, []);
  1579→    chainResidues.get(chain)!.push({ resSeq: parseInt(resSeqStr), ...entry });
  1580→  }
  1581→  for (const [, list] of chainResidues) {
  1582→    list.sort((a, b) => a.resSeq - b.resSeq);
  1583→    for (let i = 0; i < list.length - 1; i++) {
  1584→      if (list[i].c && list[i + 1].n) {
  1585→        cNPairs.push({ a: list[i].c!, b: list[i + 1].n! });
  1586→      }
  1587→    }
  1588→  }
  1589→
  1590→  return {
  1591→    nCa: computeStats(nCaPairs, 1.46), // expected N-CA ~1.46 Å
  1592→    caC: computeStats(caCPairs, 1.52), // expected CA-C ~1.52 Å
  1593→    cN: computeStats(cNPairs, 1.33),   // expected C-N (peptide) ~1.33 Å
  1594→  };
  1595→}

  1597→/**
  1598→ * Format a Smith-Waterman alignment as a text string for export.
  1599→ * Shows the two sequences with match indicators.
  1600→ */
  1601→export function formatAlignmentText(
  1602→  seq1: string,
  1603→  seq2: string,
  1604→  pairs: [number, number][]
  1605→): string {
  1606→  const lines: string[] = [];
  1607→  let line1 = "";
  1608→  let line2 = "";
  1609→  let match = "";
  1610→
  1611→  let i1 = 0;
  1612→  let i2 = 0;
  1613→  let pairIdx = 0;
  1614→
  1615→  while (i1 < seq1.length || i2 < seq2.length) {
  1616→    const nextPair = pairs[pairIdx];
  1617→    const atPair = nextPair && nextPair[0] === i1 && nextPair[1] === i2;
  1618→
  1619→    if (atPair) {
  1620→      line1 += seq1[i1];
  1621→      line2 += seq2[i2];
  1622→      match += seq1[i1] === seq2[i2] ? "|" : ".";
  1623→      i1++;
  1624→      i2++;
  1625→      pairIdx++;
  1626→    } else {
  1627→      // Gap — figure out which sequence has the gap.
  1628→      const skip1 = !nextPair || nextPair[0] > i1;
  1629→      if (skip1 && i1 < seq1.length) {
  1630→        line1 += seq1[i1];
  1631→        line2 += "-";
  1632→        match += " ";
  1633→        i1++;
  1634→      } else if (i2 < seq2.length) {
  1635→        line1 += "-";
  1636→        line2 += seq2[i2];
  1637→        match += " ";
  1638→        i2++;
  1639→      } else {
  1640→        break;
  1641→      }
  1642→    }
  1643→
  1644→    // Line break every 60 chars.
  1645→    if (line1.length >= 60) {
  1646→      lines.push(line1);
  1647→      lines.push(match);
  1648→      lines.push(line2);
  1649→      lines.push("");
  1650→      line1 = "";
  1651→      line2 = "";
  1652→      match = "";
  1653→    }
  1654→  }
  1655→
  1656→  if (line1) {
  1657→    lines.push(line1);
  1658→    lines.push(match);
  1659→    lines.push(line2);
  1660→  }
  1661→
  1662→  return lines.join("\n");
  1663→}

  1665→export interface BFactorStats {
  1666→  min: number;
  1667→  max: number;
  1668→  mean: number;
  1669→  stdDev: number;
  1670→  perResidue: { chain: string; resSeq: number; resName: string; bfactor: number; zScore: number; isOutlier: boolean }[];
  1671→  histogram: { bin: number; count: number; binLabel: string }[];
  1672→}

  1674→/**
  1675→ * Parse B-factor (temperature factor) from PDB and compute distribution stats.
  1676→ * B-factor is in columns 61-66 (right-justified, 1 decimal implied).
  1677→ */
  1678→export function computeBFactorStats(pdb: string): BFactorStats | null {
  1679→  const lines = pdb.split(/\r?\n/);
  1680→  const perResidue: BFactorStats["perResidue"] = [];
  1681→  const bfactors: number[] = [];
  1682→  const seen = new Set<string>();
  1683→
  1684→  for (const line of lines) {
  1685→    const rec = line.substring(0, 6).trim();
  1686→    if (rec !== "ATOM") continue;
  1687→    const atomName = line.substring(12, 16).trim();
  1688→    if (atomName !== "CA") continue; // use CA only for per-residue
  1689→    const resName = line.substring(17, 20).trim();
  1690→    const chain = line.substring(21, 22).trim() || " ";
  1691→    const resSeq = parseInt(line.substring(22, 26), 10);
  1692→    if (Number.isNaN(resSeq)) continue;
  1693→    const bfactor = parseFloat(line.substring(60, 66));
  1694→    if (Number.isNaN(bfactor)) continue;
  1695→
  1696→    const key = `${chain}:${resSeq}`;
  1697→    if (seen.has(key)) continue;
  1698→    seen.add(key);
  1699→
  1700→    bfactors.push(bfactor);
  1701→    perResidue.push({ chain, resSeq, resName, bfactor, zScore: 0, isOutlier: false });
  1702→  }
  1703→
  1704→  if (bfactors.length === 0) return null;
  1705→
  1706→  const n = bfactors.length;
  1707→  const mean = bfactors.reduce((s, b) => s + b, 0) / n;
  1708→  const variance = bfactors.reduce((s, b) => s + (b - mean) ** 2, 0) / n;
  1709→  const stdDev = Math.sqrt(variance);
  1710→  const min = Math.min(...bfactors);
  1711→  const max = Math.max(...bfactors);
  1712→
  1713→  // Compute z-scores and flag outliers (|z| > 2).
  1714→  for (const r of perResidue) {
  1715→    r.zScore = stdDev > 0 ? (r.bfactor - mean) / stdDev : 0;
  1716→    r.isOutlier = Math.abs(r.zScore) > 2;
  1717→  }
  1718→
  1719→  // Build histogram (10 bins from min to max).
  1720→  const numBins = 10;
  1721→  const binWidth = (max - min) / numBins || 1;
  1722→  const histogram = Array.from({ length: numBins }, (_, i) => ({
  1723→    bin: i,
  1724→    count: 0,
  1725→    binLabel: `${(min + i * binWidth).toFixed(1)}-${(min + (i + 1) * binWidth).toFixed(1)}`,
  1726→  }));
  1727→  for (const b of bfactors) {
  1728→    let binIdx = Math.floor((b - min) / binWidth);
  1729→    if (binIdx >= numBins) binIdx = numBins - 1;
  1730→    if (binIdx < 0) binIdx = 0;
  1731→    histogram[binIdx].count++;
  1732→  }
  1733→
  1734→  return { min, max, mean, stdDev, perResidue, histogram };
  1735→}

  1783→// Amino acid pKa values for ionizable sidechains.
  1784→const AA_PKA: Record<string, { group: string; pKa: number; chargeWhenProtonated: number }> = {
  1785→  ASP: { group: "sidechain-COOH", pKa: 3.65, chargeWhenProtonated: 0 },    // deprotonated = -1
  1786→  GLU: { group: "sidechain-COOH", pKa: 4.25, chargeWhenProtonated: 0 },    // deprotonated = -1
  1787→  HIS: { group: "imidazole", pKa: 6.0, chargeWhenProtonated: 1 },          // protonated = +1
  1788→  CYS: { group: "thiol", pKa: 8.3, chargeWhenProtonated: 0 },              // deprotonated = -1
  1789→  TYR: { group: "phenol", pKa: 10.07, chargeWhenProtonated: 0 },           // deprotonated = -1
  1790→  LYS: { group: "amino", pKa: 10.53, chargeWhenProtonated: 1 },            // protonated = +1
  1791→  ARG: { group: "guanidino", pKa: 12.48, chargeWhenProtonated: 1 },        // protonated = +1
  1792→};
  1793→
  1794→// Terminal group pKa values.
  1795→const N_TERM_PKA = 9.0;  // protonated = +1
  1796→const C_TERM_PKA = 2.0;  // protonated = 0, deprotonated = -1
  1797→
  1798→/**
  1799→ * Compute per-residue charge at a given pH using Henderson-Hasselbalch.
  1800→ * For acidic groups (COOH): charge = -1 / (1 + 10^(pKa - pH))
  1801→ * For basic groups (NH): charge = +1 / (1 + 10^(pH - pKa))
  1802→ *
  1803→ * Includes N-terminal (+1 when protonated) and C-terminal (-1 when deprotonated)
  1804→ * contributions applied to the first and last residues of each chain.
  1805→ */
  1806→export function computeChargeAtPH(
  1807→  pdb: string,
  1808→  pH: number
  1809→): ChargeInfo & { pH: number; nTermCharge: number; cTermCharge: number } {
  1810→  const parsed = parsePdb(pdb);
  1811→  const ca = parsed.ca;
  1812→
  1813→  // Group by chain to find termini.
  1814→  const chainResidues = new Map<string, typeof ca>();
  1815→  for (const a of ca) {
  1816→    if (!chainResidues.has(a.chain)) chainResidues.set(a.chain, []);
  1817→    chainResidues.get(a.chain)!.push(a);
  1818→  }
  1819→
  1820→  // Determine which residues are N-term / C-term (first/last per chain by resSeq).
  1821→  const nTermKeys = new Set<string>();
  1822→  const cTermKeys = new Set<string>();
  1823→  for (const [, residues] of chainResidues) {
  1824→    const sorted = [...residues].sort((a, b) => a.resSeq - b.resSeq);
  1825→    if (sorted.length > 0) {
  1826→      nTermKeys.add(`${sorted[0].chain}:${sorted[0].resSeq}`);
  1827→      cTermKeys.add(`${sorted[sorted.length - 1].chain}:${sorted[sorted.length - 1].resSeq}`);
  1828→    }
  1829→  }
  1830→
  1831→  let nTermCharge = 0;
  1832→  let cTermCharge = 0;
  1833→  const perResidue = ca.map((a) => {
  1834→    let charge = 0;
  1835→    const pkaInfo = AA_PKA[a.resName];
  1836→    if (pkaInfo) {
  1837→      if (pkaInfo.chargeWhenProtonated > 0) {
  1838→        // Basic group (protonated = positive).
  1839→        charge = pkaInfo.chargeWhenProtonated / (1 + Math.pow(10, pH - pkaInfo.pKa));
  1840→      } else {
  1841→        // Acidic group (deprotonated = negative).
  1842→        charge = -1 / (1 + Math.pow(10, pkaInfo.pKa - pH));
  1843→      }
  1844→    }
  1845→
  1846→    const key = `${a.chain}:${a.resSeq}`;
  1847→    if (nTermKeys.has(key)) {
  1848→      const nCharge = 1 / (1 + Math.pow(10, pH - N_TERM_PKA));
  1849→      charge += nCharge;
  1850→      nTermCharge += nCharge;
  1851→    }
  1852→    if (cTermKeys.has(key)) {
  1853→      const cCharge = -1 / (1 + Math.pow(10, C_TERM_PKA - pH));
  1854→      charge += cCharge;
  1855→      cTermCharge += cCharge;
  1856→    }
  1857→
  1858→    return {
  1859→      chain: a.chain,
  1860→      resSeq: a.resSeq,
  1861→      resName: a.resName,
  1862→      charge,
  1863→    };
  1864→  });
  1865→
  1866→  const totalCharge = perResidue.reduce((s, r) => s + r.charge, 0);
  1867→  const positiveCount = perResidue.filter((r) => r.charge > 0.1).length;
  1868→  const negativeCount = perResidue.filter((r) => r.charge < -0.1).length;
  1869→
  1870→  return {
  1871→    totalCharge,
  1872→    positiveCount,
  1873→    negativeCount,
  1874→    perResidue,
  1875→    pH,
  1876→    nTermCharge,
  1877→    cTermCharge,
  1878→  };
  1879→}

  1881→/**
  1882→ * Compute the isoelectric point (pI) — the pH where net charge = 0.
  1883→ * Uses bisection search over pH 0-14.
  1884→ */
