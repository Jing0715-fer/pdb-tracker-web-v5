     1→     1→"use client";
     2→     2→
     3→     3→/**
     4→     4→ * Executor that turns parsed LlmCommand objects into real Molstar API calls
     5→     5→ * OR backend analysis API calls (for analyze_* commands).
     6→     6→ *
     7→     7→ * For analyze_* commands, we return the fetched data so the chat panel can
     8→     8→ * feed it back to the LLM in a follow-up turn (ReAct pattern).
     9→     9→ */
    10→    10→
    11→    11→import type { LlmCommand, ResidueRef } from "./command-schema";
    12→    12→import type { MolstarPlugin, MolstarViewer } from "./types";
    13→    13→import { useAppStore, type ElectrostaticViz, type DruggabilityViz, type ScreeningViz, type PocketDetectionViz } from "./store";
    14→    14→
    15→    15→export interface CommandResult {
    16→    16→  ok: boolean;
    17→    17→  detail?: string;
    18→    18→  /** Optional data to return to the UI (e.g. measurement values, analysis results). */
    19→    19→  data?: Record<string, unknown>;
    20→    20→  /** For analyze_* commands: the raw analysis result to feed back to the LLM. */
    21→    21→  analysisResult?: unknown;
    22→    22→}
    23→    23→
    24→    24→// ============================================================
    25→    25→// Backend analysis API helpers (called from the browser)
    26→    26→// ============================================================
    27→    27→
    28→    28→async function fetchMetadata(id: string, includeInterfaces: boolean) {
    29→    29→  const url = `/api/analyze/metadata?id=${encodeURIComponent(id)}&interfaces=${includeInterfaces ? 1 : 0}&format=markdown`;
    30→    30→  const res = await fetch(url);
    31→    31→  if (!res.ok) {
    32→    32→    const err = await res.json().catch(() => ({}));
    33→    33→    throw new Error(err.detail || `HTTP ${res.status}`);
    34→    34→  }
    35→    35→  return await res.text();
    36→    36→}
    37→    37→
    38→    38→async function fetchInterface(id: string, assembly: number) {
    39→    39→  const url = `/api/analyze/interface?id=${encodeURIComponent(id)}&assembly=${assembly}`;
    40→    40→  const res = await fetch(url);
    41→    41→  if (!res.ok) {
    42→    42→    const err = await res.json().catch(() => ({}));
    43→    43→    throw new Error(err.detail || `HTTP ${res.status}`);
    44→    44→  }
    45→    45→  return await res.json();
    46→    46→}
    47→    47→
    48→    48→async function fetchCliList() {
    49→    49→  const res = await fetch("/api/cli/list");
    50→    50→  if (!res.ok) throw new Error(`HTTP ${res.status}`);
    51→    51→  return await res.json();
    52→    52→}
    53→    53→
    54→    54→async function runRecipe(
    55→    55→  recipe: string,
    56→    56→  pdbId?: string,
    57→    57→  params?: Record<string, unknown>
    58→    58→) {
    59→    59→  const res = await fetch("/api/analyze/run", {
    60→    60→    method: "POST",
    61→    61→    headers: { "Content-Type": "application/json" },
    62→    62→    body: JSON.stringify({ recipe, pdbId, params }),
    63→    63→  });
    64→    64→  if (!res.ok) {
    65→    65→    const err = await res.json().catch(() => ({}));
    66→    66→    throw new Error(err.detail || err.error || `HTTP ${res.status}`);
    67→    67→  }
    68→    68→  return await res.json();
    69→    69→}
    70→    70→
    71→    71→export async function executeCommand(
    72→    72→  viewer: MolstarViewer,
    73→    73→  cmd: LlmCommand
    74→    74→): Promise<CommandResult> {
    75→    75→  const plugin = viewer.plugin;
    76→    76→  try {
    77→    77→    switch (cmd.type) {
    78→    78→      // ---------- Real analysis (NEW) ----------
    79→    79→      case "analyze_metadata": {
    80→    80→        const md = await fetchMetadata(cmd.id, cmd.includeInterfaces ?? true);
    81→    81→        return {
    82→    82→          ok: true,
    83→    83→          detail: `Fetched RCSB metadata for ${cmd.id} (${md.length} chars)`,
    84→    84→          analysisResult: { kind: "metadata", id: cmd.id, markdown: md },
    85→    85→        };
    86→    86→      }
    87→    87→      case "analyze_interface": {
    88→    88→        const data = await fetchInterface(cmd.id, cmd.assembly ?? 1);
    89→    89→        return {
    90→    90→          ok: true,
    91→    91→          detail: `Fetched interface data for ${cmd.id} assembly ${cmd.assembly ?? 1}`,
    92→    92→          analysisResult: { kind: "interface", id: cmd.id, data },
    93→    93→        };
    94→    94→      }
    95→    95→      case "analyze_cli_list": {
    96→    96→        const data = await fetchCliList();
    97→    97→        const avail = data.clis?.filter((c: any) => c.available) ?? [];
    98→    98→        return {
    99→    99→          ok: true,
   100→   100→          detail: `Local CLIs: ${avail.map((c: any) => c.id).join(", ") || "none"}`,
   101→   101→          analysisResult: { kind: "cli_list", data },
   102→   102→        };
   103→   103→      }
   104→   104→      case "analyze_run": {
   105→   105→        const data = await runRecipe(cmd.recipe, cmd.pdbId, cmd.params);
   106→   106→        return {
   107→   107→          ok: true,
   108→   108→          detail: `Recipe ${cmd.recipe} ok: ${data.data ? "with data" : "see stdout"}`,
   109→   109→          analysisResult: { kind: "recipe", recipe: cmd.recipe, data },
   110→   110→        };
   111→   111→      }
   112→   112→
   113→   113→      // ---------- Loading ----------
   114→   114→      case "load_pdb":
   115→   115→        await viewer.loadPdb(cmd.id);
   116→   116→        return { ok: true, detail: `Loaded PDB ${cmd.id}` };
   117→   117→
   118→   118→      case "load_alphafold":
   119→   119→        await viewer.loadAlphaFoldDb(cmd.uniprotId);
   120→   120→        return { ok: true, detail: `Loaded AlphaFold ${cmd.uniprotId}` };
   121→   121→
   122→   122→      case "load_emdb":
   123→   123→        await viewer.loadEmdb(cmd.emdbId, { detail: cmd.detail ?? 3 });
   124→   124→        return { ok: true, detail: `Loaded EMDB ${cmd.emdbId}` };
   125→   125→
   126→   126→      case "load_structure_url":
   127→   127→        await viewer.loadStructureFromUrl(
   128→   128→          cmd.url,
   129→   129→          cmd.format ?? "mmcif",
   130→   130→          cmd.isBinary ?? false
   131→   131→        );
   132→   132→        return { ok: true, detail: `Loaded structure from ${cmd.url}` };
   133→   133→
   134→   134→      case "load_volume_url": {
   135→   135→        const color = cmd.color
   136→   136→          ? hexToNumber(cmd.color)
   137→   137→          : 0x3377aa;
   138→   138→        await viewer.loadVolumeFromUrl(
   139→   139→          { url: cmd.url, format: cmd.format, isBinary: cmd.isBinary },
   140→   140→          [{ type: "relative", value: cmd.isoValue, color }],
   141→   141→          { isLazy: true }
   142→   142→        );
   143→   143→        return { ok: true, detail: `Loaded volume @ iso ${cmd.isoValue}` };
   144→   144→      }
   145→   145→
   146→   146→      // ---------- Representation ----------
   147→   147→      case "set_representation": {
   148→   148→        const structures = getStructures(plugin, cmd.structures);
   149→   149→        if (structures.length === 0)
   150→   150→          return { ok: false, detail: "No structures loaded" };
   151→   151→        await plugin.managers.structure.component.applyPreset(
   152→   152→          structures,
   153→   153→          cmd.preset
   154→   154→        );
   155→   155→        return { ok: true, detail: `Applied preset: ${cmd.preset}` };
   156→   156→      }
   157→   157→
   158→   158→      case "set_color_theme": {
   159→   159→        const structures = getStructures(plugin, cmd.structures);
   160→   160→        const components = collectComponents(plugin, structures);
   161→   161→        if (components.length === 0)
   162→   162→          return { ok: false, detail: "No components to color" };
   163→   163→        plugin.managers.structure.component.updateRepresentationsTheme(
   164→   164→          components,
   165→   165→          { color: cmd.theme }
   166→   166→        );
   167→   167→        return { ok: true, detail: `Color theme: ${cmd.theme}` };
   168→   168→      }
   169→   169→
   170→   170→      case "set_uniform_color": {
   171→   171→        const structures = getStructures(plugin, cmd.structures);
   172→   172→        const components = collectComponents(plugin, structures);
   173→   173→        if (components.length === 0)
   174→   174→          return { ok: false, detail: "No components to color" };
   175→   175→        plugin.managers.structure.component.updateRepresentationsTheme(
   176→   176→          components,
   177→   177→          { color: "uniform", colorParams: { value: hexToNumber(cmd.color) } }
   178→   178→        );
   179→   179→        return { ok: true, detail: `Uniform color: ${cmd.color}` };
   180→   180→      }
   181→   181→
   182→   182→      // ---------- Camera ----------
   183→   183→      case "focus_residue": {
   184→   184→        const loci = await lociFromResidue(viewer, {
   185→   185→          chain: cmd.chain,
   186→   186→          resno: cmd.resno,
   187→   187→          compId: cmd.compId,
   188→   188→        });
   189→   189→        if (!loci) return { ok: false, detail: "Residue not found" };
   190→   190→        plugin.managers.camera.focusLoci(loci, { minRadius: 12 });
   191→   191→        return { ok: true, detail: `Focused residue ${cmd.chain ?? ""}:${cmd.resno ?? cmd.compId}` };
   192→   192→      }
   193→   193→
   194→   194→      case "focus_ligand": {
   195→   195→        const loci = await lociFromResidue(viewer, { compId: cmd.compId });
   196→   196→        if (!loci) return { ok: false, detail: `Ligand ${cmd.compId} not found` };
   197→   197→        plugin.managers.camera.focusLoci(loci, { minRadius: 10 });
   198→   198→        return { ok: true, detail: `Focused ligand ${cmd.compId}` };
   199→   199→      }
   200→   200→
   201→   201→      case "focus_chain": {
   202→   202→        const loci = await lociFromChain(viewer, cmd.chain);
   203→   203→        if (!loci) return { ok: false, detail: `Chain ${cmd.chain} not found` };
   204→   204→        plugin.managers.camera.focusLoci(loci);
   205→   205→        return { ok: true, detail: `Focused chain ${cmd.chain}` };
   206→   206→      }
   207→   207→
   208→   208→      case "focus_selection": {
   209→   209→        const sel = plugin.managers.structure.selection.getBoundary();
   210→   210→        if (sel?.sphere) {
   211→   211→          plugin.managers.camera.focusSphere({
   212→   212→            center: sel.sphere.center,
   213→   213→            radius: (sel.sphere.radius ?? 15) + 5,
   214→   214→          });
   215→   215→          return { ok: true, detail: "Focused selection" };
   216→   216→        }
   217→   217→        return { ok: false, detail: "No selection to focus" };
   218→   218→      }
   219→   219→
   220→   220→      case "reset_camera":
   221→   221→        plugin.managers.camera.reset();
   222→   222→        return { ok: true, detail: "Camera reset" };
   223→   223→
   224→   224→      // ---------- Measurements ----------
   225→   225→      case "measure_distance": {
   226→   226→        const a = await lociFromResidue(viewer, cmd.a, cmd.atomA);
   227→   227→        const b = await lociFromResidue(viewer, cmd.b, cmd.atomB);
   228→   228→        if (!a || !b)
   229→   229→          return { ok: false, detail: "Could not resolve both residues" };
   230→   230→        await plugin.managers.structure.measurement.addDistance(a, b);
   231→   231→        return { ok: true, detail: "Distance measurement added" };
   232→   232→      }
   233→   233→
   234→   234→      case "measure_angle": {
   235→   235→        const a = await lociFromResidue(viewer, cmd.a);
   236→   236→        const b = await lociFromResidue(viewer, cmd.b);
   237→   237→        const c = await lociFromResidue(viewer, cmd.c);
   238→   238→        if (!a || !b || !c)
   239→   239→          return { ok: false, detail: "Could not resolve all 3 residues" };
   240→   240→        await plugin.managers.structure.measurement.addAngle(a, b, c);
   241→   241→        return { ok: true, detail: "Angle measurement added" };
   242→   242→      }
   243→   243→
   244→   244→      case "measure_dihedral": {
   245→   245→        const a = await lociFromResidue(viewer, cmd.a);
   246→   246→        const b = await lociFromResidue(viewer, cmd.b);
   247→   247→        const c = await lociFromResidue(viewer, cmd.c);
   248→   248→        const d = await lociFromResidue(viewer, cmd.d);
   249→   249→        if (!a || !b || !c || !d)
   250→   250→          return { ok: false, detail: "Could not resolve all 4 residues" };
   251→   251→        await plugin.managers.structure.measurement.addDihedral(a, b, c, d);
   252→   252→        return { ok: true, detail: "Dihedral measurement added" };
   253→   253→      }
   254→   254→
   255→   255→      case "label_residue": {
   256→   256→        const loci = await lociFromResidue(viewer, cmd.target);
   257→   257→        if (!loci) return { ok: false, detail: "Residue not found" };
   258→   258→        await plugin.managers.structure.measurement.addLabel(loci, {
   259→   259→          customText: cmd.text ?? "",
   260→   260→        });
   261→   261→        return { ok: true, detail: "Label added" };
   262→   262→      }
   263→   263→
   264→   264→      case "clear_measurements":
   265→   265→        plugin.managers.structure.measurement.clear();
   266→   266→        return { ok: true, detail: "Measurements cleared" };
   267→   267→
   268→   268→      // ---------- Interactions ----------
   269→   269→      case "show_interactions": {
   270→   270→        const targetLoci = await resolveInteractionsTarget(viewer, cmd.target);
   271→   271→        if (!targetLoci)
   272→   272→          return { ok: false, detail: "Could not resolve interactions target" };
   273→   273→        await showInteractionsAround(plugin, targetLoci, cmd.radius ?? 8);
   274→   274→        return { ok: true, detail: `Interactions within ${cmd.radius ?? 8} Å` };
   275→   275→      }
   276→   276→
   277→   277→      case "clear_interactions":
   278→   278→        await clearInteractions(plugin);
   279→   279→        return { ok: true, detail: "Interactions cleared" };
   280→   280→
   281→   281→      // ---------- Selection ----------
   282→   282→      case "select": {
   283→   283→        let loci: unknown = null;
   284→   284→        if (cmd.target === "all") {
   285→   285→          loci = getFirstStructureData(plugin);
   286→   286→        } else if (cmd.target === "ligand") {
   287→   287→          loci = await lociFromResidue(viewer, {});
   288→   288→        } else {
   289→   289→          loci = await lociFromResidue(viewer, cmd.target);
   290→   290→        }
   291→   291→        if (!loci) return { ok: false, detail: "Selection target not found" };
   292→   292→        const action = cmd.action ?? "set";
   293→   293→        // @ts-expect-error: action is union-typed at runtime
   294→   294→        plugin.managers.structure.selection.modify(action, loci);
   295→   295→        return { ok: true, detail: `Selected (${cmd.action ?? "set"})` };
   296→   296→      }
   297→   297→
   298→   298→      case "clear_selection":
   299→   299→        plugin.managers.structure.selection.clear();
   300→   300→        plugin.managers.interactivity.lociSelects.deselectAll();
   301→   301→        return { ok: true, detail: "Selection cleared" };
   302→   302→
   303→   303→      case "set_granularity":
   304→   304→        plugin.managers.interactivity.setProps({ granularity: cmd.granularity });
   305→   305→        return { ok: true, detail: `Granularity: ${cmd.granularity}` };
   306→   306→
   307→   307→      // ---------- Animation ----------
   308→   308→      case "toggle_spin": {
   309→   309→        setTrackballAnimate(plugin, "spin", { speed: cmd.speed ?? 0.1 });
   310→   310→        return { ok: true, detail: "Spin animation enabled" };
   311→   311→      }
   312→   312→      case "toggle_rock": {
   313→   313→        setTrackballAnimate(plugin, "rock", {});
   314→   314→        return { ok: true, detail: "Rock animation enabled" };
   315→   315→      }
   316→   316→      case "stop_animation": {
   317→   317→        setTrackballAnimate(plugin, undefined, {});
   318→   318→        return { ok: true, detail: "Animation stopped" };
   319→   319→      }
   320→   320→
   321→   321→      // ---------- Export ----------
   322→   322→      case "export_snapshot": {
   323→   323→        const dataUri = await plugin.helpers?.viewportScreenshot?.getImageDataUri(
   324→   324→          {
   325→   325→            width: cmd.width,
   326→   326→            height: cmd.height,
   327→   327→            transparency: false,
   328→   328→            axes: true,
   329→   329→          }
   330→   330→        );
   331→   331→        if (!dataUri)
   332→   332→          return { ok: false, detail: "Screenshot failed" };
   333→   333→        const a = document.createElement("a");
   334→   334→        a.href = dataUri;
   335→   335→        a.download = `molstar-snapshot-${Date.now()}.png`;
   336→   336→        a.click();
   337→   337→        return { ok: true, detail: "Snapshot exported", data: { dataUri } };
   338→   338→      }
   339→   339→
   340→   340→      // ---------- Capture (returns data URI for embedding, no download) ----------
   341→   341→      case "capture_snapshot": {
   342→   342→        // Step 1: apply camera angle if requested
   343→   343→        if (cmd.angle) {
   344→   344→          try {
   345→   345→            await applyCameraAngle(plugin, cmd.angle);
   346→   346→          } catch (err) {
   347→   347→            console.warn("[capture_snapshot] angle adjust failed:", err);
   348→   348→          }
   349→   349→        }
   350→   350→        // Step 2: add text labels at residue positions
   351→   351→        if (Array.isArray(cmd.labels) && cmd.labels.length > 0) {
   352→   352→          for (const lbl of cmd.labels) {
   353→   353→            try {
   354→   354→              if (lbl.chain !== undefined && lbl.resno !== undefined) {
   355→   355→                const loci = await lociFromResidue(viewer, {
   356→   356→                  chain: lbl.chain,
   357→   357→                  resno: lbl.resno,
   358→   358→                });
   359→   359→                if (loci) {
   360→   360→                  await plugin.managers.structure.measurement.addLabel(loci, {
   361→   361→                    customText: lbl.text ?? "",
   362→   362→                  });
   363→   363→                }
   364→   364→              }
   365→   365→            } catch (err) {
   366→   366→              console.warn(
   367→   367→                `[capture_snapshot] label "${lbl.text}" failed:`,
   368→   368→                err
   369→   369→              );
   370→   370→            }
   371→   371→          }
   372→   372→          // Allow labels to render before capture
   373→   373→          await new Promise((r) => setTimeout(r, 100));
   374→   374→        }
   375→   375→        // Step 3: capture
   376→   376→        const dataUri =
   377→   377→          await plugin.helpers?.viewportScreenshot?.getImageDataUri({
   378→   378→            width: cmd.width ?? 1600,
   379→   379→            height: cmd.height ?? 900,
   380→   380→            transparency: false,
   381→   381→            axes: true,
   382→   382→          });
   383→   383→        if (!dataUri)
   384→   384→          return { ok: false, detail: "Capture failed" };
   385→   385→        return {
   386→   386→          ok: true,
   387→   387→          detail: `Captured${cmd.label ? `: ${cmd.label}` : ""}${
   388→   388→            cmd.angle ? ` (${cmd.angle})` : ""
   389→   389→          }`,
   390→   390→          data: {
   391→   391→            dataUri,
   392→   392→            label: cmd.label ?? "",
   393→   393→            angle: cmd.angle ?? "",
   394→   394→          },
   395→   395→        };
   396→   396→      }
   397→   397→
   398→   398→      // ---------- Alignment ----------
   399→   399→      case "align_structures": {
   400→   400→        const result = await alignStructures(
   401→   401→          plugin,
   402→   402→          cmd.ref,
   403→   403→          cmd.mobile,
   404→   404→          cmd.method ?? "tm-align"
   405→   405→        );
   406→   406→        return result;
   407→   407→      }
   408→   408→
   409→   409→      // ---------- Misc ----------
   410→   410→      case "set_background": {
   411→   411→        const canvas3d = plugin.canvas3d as
   412→   412→          | { setProps?: (fn: (p: unknown) => void) => void }
   413→   413→          | undefined;
   414→   414→        if (canvas3d?.setProps) {
   415→   415→          canvas3d.setProps((p: unknown) => {
   416→   416→            const props = p as { renderer?: { backgroundColor?: unknown } };
   417→   417→            props.renderer = props.renderer ?? {};
   418→   418→            props.renderer.backgroundColor = hexToNumber(cmd.color);
   419→   419→          });
   420→   420→        }
   421→   421→        return { ok: true, detail: `Background: ${cmd.color}` };
   422→   422→      }
   423→   423→
   424→   424→      case "toggle_component_visibility":
   425→   425→        return { ok: true, detail: `Toggled ${cmd.component} (no-op stub)` };
   426→   426→
   427→   427→      // ---------- APBS Electrostatic 3D Visualization ----------
   428→   428→      case "show_electrostatic_surface": {
   429→   429→        const params: Record<string, unknown> = {};
   430→   430→        if (cmd.chain) params.chain = cmd.chain;
   431→   431→        if (cmd.ionicStrength) params.ionic_strength = cmd.ionicStrength;
   432→   432→        const pdbId = cmd.pdbId ?? useAppStore.getState().structures[0]?.id ?? "";
   433→   433→        const data = await runRecipe("apbs_electrostatic", pdbId || undefined, params);
   434→   434→        const d = data?.data;
   435→   435→        if (!d || d.error) {
   436→   436→          return { ok: false, detail: `APBS failed: ${d?.error ?? "no data"}` };
   437→   437→        }
   438→   438→        // Apply molecular-surface + partial-charge color
   439→   439→        const structs = plugin.managers.structure.hierarchy.current.structures;
   440→   440→        if (structs.length > 0) {
   441→   441→          try {
   442→   442→            plugin.managers.structure.component.applyPreset([structs[0]], "molecular-surface");
   443→   443→            await new Promise((r) => setTimeout(r, 200));
   444→   444→            const comps = structs[0]?.components ?? [];
   445→   445→            if (comps.length > 0) {
   446→   446→              plugin.managers.structure.component.updateRepresentationsTheme(comps, {
   447→   447→                color: "partial-charge",
   448→   448→              });
   449→   449→            }
   450→   450→          } catch (e) {
   451→   451→            console.warn("[show_electrostatic_surface] representation error:", e);
   452→   452→          }
   453→   453→        }
   454→   454→        const viz: ElectrostaticViz = {
   455→   455→          pdbId: pdbId || "unknown",
