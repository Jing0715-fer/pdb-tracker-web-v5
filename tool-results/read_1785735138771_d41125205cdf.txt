     1→"use client";
     2→
     3→/**
     4→ * Executor that turns parsed LlmCommand objects into real Molstar API calls
     5→ * OR backend analysis API calls (for analyze_* commands).
     6→ *
     7→ * For analyze_* commands, we return the fetched data so the chat panel can
     8→ * feed it back to the LLM in a follow-up turn (ReAct pattern).
     9→ */
    10→
    11→import type { LlmCommand, ResidueRef } from "@/lib/llm/command-schema";
    12→import type { MolstarPlugin, MolstarViewer } from "./types";
    13→import { useAppStore, type ElectrostaticViz, type DruggabilityViz, type ScreeningViz, type PocketDetectionViz } from "@/lib/store";
    14→
    15→export interface CommandResult {
    16→  ok: boolean;
    17→  detail?: string;
    18→  /** Optional data to return to the UI (e.g. measurement values, analysis results). */
    19→  data?: Record<string, unknown>;
    20→  /** For analyze_* commands: the raw analysis result to feed back to the LLM. */
    21→  analysisResult?: unknown;
    22→}
    23→
    24→// ============================================================
    25→// Backend analysis API helpers (called from the browser)
    26→// ============================================================
    27→
    28→async function fetchMetadata(id: string, includeInterfaces: boolean) {
    29→  const url = `/api/analyze/metadata?id=${encodeURIComponent(id)}&interfaces=${includeInterfaces ? 1 : 0}&format=markdown`;
    30→  const res = await fetch(url);
    31→  if (!res.ok) {
    32→    const err = await res.json().catch(() => ({}));
    33→    throw new Error(err.detail || `HTTP ${res.status}`);
    34→  }
    35→  return await res.text();
    36→}
    37→
    38→async function fetchInterface(id: string, assembly: number) {
    39→  const url = `/api/analyze/interface?id=${encodeURIComponent(id)}&assembly=${assembly}`;
    40→  const res = await fetch(url);
    41→  if (!res.ok) {
    42→    const err = await res.json().catch(() => ({}));
    43→    throw new Error(err.detail || `HTTP ${res.status}`);
    44→  }
    45→  return await res.json();
    46→}
    47→
    48→async function fetchCliList() {
    49→  const res = await fetch("/api/cli/list");
    50→  if (!res.ok) throw new Error(`HTTP ${res.status}`);
    51→  return await res.json();
    52→}
    53→
    54→async function runRecipe(
    55→  recipe: string,
    56→  pdbId?: string,
    57→  params?: Record<string, unknown>
    58→) {
    59→  const res = await fetch("/api/analyze/run", {
    60→    method: "POST",
    61→    headers: { "Content-Type": "application/json" },
    62→    body: JSON.stringify({ recipe, pdbId, params }),
    63→  });
    64→  if (!res.ok) {
    65→    const err = await res.json().catch(() => ({}));
    66→    throw new Error(err.detail || err.error || `HTTP ${res.status}`);
    67→  }
    68→  return await res.json();
    69→}
    70→
    71→export async function executeCommand(
    72→  viewer: MolstarViewer,
    73→  cmd: LlmCommand
    74→): Promise<CommandResult> {
    75→  const plugin = viewer.plugin;
    76→  try {
    77→    switch (cmd.type) {
    78→      // ---------- Real analysis (NEW) ----------
    79→      case "analyze_metadata": {
    80→        const md = await fetchMetadata(cmd.id, cmd.includeInterfaces ?? true);
    81→        return {
    82→          ok: true,
    83→          detail: `Fetched RCSB metadata for ${cmd.id} (${md.length} chars)`,
    84→          analysisResult: { kind: "metadata", id: cmd.id, markdown: md },
    85→        };
    86→      }
    87→      case "analyze_interface": {
    88→        const data = await fetchInterface(cmd.id, cmd.assembly ?? 1);
    89→        return {
    90→          ok: true,
    91→          detail: `Fetched interface data for ${cmd.id} assembly ${cmd.assembly ?? 1}`,
    92→          analysisResult: { kind: "interface", id: cmd.id, data },
    93→        };
    94→      }
    95→      case "analyze_cli_list": {
    96→        const data = await fetchCliList();
    97→        const avail = data.clis?.filter((c: any) => c.available) ?? [];
    98→        return {
    99→          ok: true,
   100→          detail: `Local CLIs: ${avail.map((c: any) => c.id).join(", ") || "none"}`,
   101→          analysisResult: { kind: "cli_list", data },
   102→        };
   103→      }
   104→      case "analyze_run": {
   105→        const data = await runRecipe(cmd.recipe, cmd.pdbId, cmd.params);
   106→        return {
   107→          ok: true,
   108→          detail: `Recipe ${cmd.recipe} ok: ${data.data ? "with data" : "see stdout"}`,
   109→          analysisResult: { kind: "recipe", recipe: cmd.recipe, data },
   110→        };
   111→      }
   112→
   113→      // ---------- Loading ----------
   114→      case "load_pdb":
   115→        await viewer.loadPdb(cmd.id);
   116→        return { ok: true, detail: `Loaded PDB ${cmd.id}` };
   117→
   118→      case "load_alphafold":
   119→        await viewer.loadAlphaFoldDb(cmd.uniprotId);
   120→        return { ok: true, detail: `Loaded AlphaFold ${cmd.uniprotId}` };
   121→
   122→      case "load_emdb":
   123→        await viewer.loadEmdb(cmd.emdbId, { detail: cmd.detail ?? 3 });
   124→        return { ok: true, detail: `Loaded EMDB ${cmd.emdbId}` };
   125→
   126→      case "load_structure_url":
   127→        await viewer.loadStructureFromUrl(
   128→          cmd.url,
   129→          cmd.format ?? "mmcif",
   130→          cmd.isBinary ?? false
   131→        );
   132→        return { ok: true, detail: `Loaded structure from ${cmd.url}` };
   133→
   134→      case "load_volume_url": {
   135→        const color = cmd.color
   136→          ? hexToNumber(cmd.color)
   137→          : 0x3377aa;
   138→        await viewer.loadVolumeFromUrl(
   139→          { url: cmd.url, format: cmd.format, isBinary: cmd.isBinary },
   140→          [{ type: "relative", value: cmd.isoValue, color }],
   141→          { isLazy: true }
   142→        );
   143→        return { ok: true, detail: `Loaded volume @ iso ${cmd.isoValue}` };
   144→      }
   145→
   146→      // ---------- Representation ----------
   147→      case "set_representation": {
   148→        const structures = getStructures(plugin, cmd.structures);
   149→        if (structures.length === 0)
   150→          return { ok: false, detail: "No structures loaded" };
   151→        await plugin.managers.structure.component.applyPreset(
   152→          structures,
   153→          cmd.preset
   154→        );
   155→        return { ok: true, detail: `Applied preset: ${cmd.preset}` };
   156→      }
   157→
   158→      case "set_color_theme": {
   159→        const structures = getStructures(plugin, cmd.structures);
   160→        const components = collectComponents(plugin, structures);
   161→        if (components.length === 0)
   162→          return { ok: false, detail: "No components to color" };
   163→        plugin.managers.structure.component.updateRepresentationsTheme(
   164→          components,
   165→          { color: cmd.theme }
   166→        );
   167→        return { ok: true, detail: `Color theme: ${cmd.theme}` };
   168→      }
   169→
   170→      case "set_uniform_color": {
   171→        const structures = getStructures(plugin, cmd.structures);
   172→        const components = collectComponents(plugin, structures);
   173→        if (components.length === 0)
   174→          return { ok: false, detail: "No components to color" };
   175→        plugin.managers.structure.component.updateRepresentationsTheme(
   176→          components,
   177→          { color: "uniform", colorParams: { value: hexToNumber(cmd.color) } }
   178→        );
   179→        return { ok: true, detail: `Uniform color: ${cmd.color}` };
   180→      }
   181→
   182→      // ---------- Camera ----------
   183→      case "focus_residue": {
   184→        const loci = await lociFromResidue(viewer, {
   185→          chain: cmd.chain,
   186→          resno: cmd.resno,
   187→          compId: cmd.compId,
   188→        });
   189→        if (!loci) return { ok: false, detail: "Residue not found" };
   190→        plugin.managers.camera.focusLoci(loci, { minRadius: 12 });
   191→        return { ok: true, detail: `Focused residue ${cmd.chain ?? ""}:${cmd.resno ?? cmd.compId}` };
   192→      }
   193→
   194→      case "focus_ligand": {
   195→        const loci = await lociFromResidue(viewer, { compId: cmd.compId });
   196→        if (!loci) return { ok: false, detail: `Ligand ${cmd.compId} not found` };
   197→        plugin.managers.camera.focusLoci(loci, { minRadius: 10 });
   198→        return { ok: true, detail: `Focused ligand ${cmd.compId}` };
   199→      }
   200→
   201→      case "focus_chain": {
   202→        const loci = await lociFromChain(viewer, cmd.chain);
   203→        if (!loci) return { ok: false, detail: `Chain ${cmd.chain} not found` };
   204→        plugin.managers.camera.focusLoci(loci);
   205→        return { ok: true, detail: `Focused chain ${cmd.chain}` };
   206→      }
   207→
   208→      case "focus_selection": {
   209→        const sel = plugin.managers.structure.selection.getBoundary();
   210→        if (sel?.sphere) {
   211→          plugin.managers.camera.focusSphere({
   212→            center: sel.sphere.center,
   213→            radius: (sel.sphere.radius ?? 15) + 5,
   214→          });
   215→          return { ok: true, detail: "Focused selection" };
   216→        }
   217→        return { ok: false, detail: "No selection to focus" };
   218→      }
   219→
   220→      case "reset_camera":
   221→        plugin.managers.camera.reset();
   222→        return { ok: true, detail: "Camera reset" };
   223→
   224→      // ---------- Measurements ----------
   225→      case "measure_distance": {
   226→        const a = await lociFromResidue(viewer, cmd.a, cmd.atomA);
   227→        const b = await lociFromResidue(viewer, cmd.b, cmd.atomB);
   228→        if (!a || !b)
   229→          return { ok: false, detail: "Could not resolve both residues" };
   230→        await plugin.managers.structure.measurement.addDistance(a, b);
   231→        return { ok: true, detail: "Distance measurement added" };
   232→      }
   233→
   234→      case "measure_angle": {
   235→        const a = await lociFromResidue(viewer, cmd.a);
   236→        const b = await lociFromResidue(viewer, cmd.b);
   237→        const c = await lociFromResidue(viewer, cmd.c);
   238→        if (!a || !b || !c)
   239→          return { ok: false, detail: "Could not resolve all 3 residues" };
   240→        await plugin.managers.structure.measurement.addAngle(a, b, c);
   241→        return { ok: true, detail: "Angle measurement added" };
   242→      }
   243→
   244→      case "measure_dihedral": {
   245→        const a = await lociFromResidue(viewer, cmd.a);
   246→        const b = await lociFromResidue(viewer, cmd.b);
   247→        const c = await lociFromResidue(viewer, cmd.c);
   248→        const d = await lociFromResidue(viewer, cmd.d);
   249→        if (!a || !b || !c || !d)
   250→          return { ok: false, detail: "Could not resolve all 4 residues" };
   251→        await plugin.managers.structure.measurement.addDihedral(a, b, c, d);
   252→        return { ok: true, detail: "Dihedral measurement added" };
   253→      }
   254→
   255→      case "label_residue": {
   256→        const loci = await lociFromResidue(viewer, cmd.target);
   257→        if (!loci) return { ok: false, detail: "Residue not found" };
   258→        await plugin.managers.structure.measurement.addLabel(loci, {
   259→          customText: cmd.text ?? "",
   260→        });
   261→        return { ok: true, detail: "Label added" };
   262→      }
   263→
   264→      case "clear_measurements":
   265→        plugin.managers.structure.measurement.clear();
   266→        return { ok: true, detail: "Measurements cleared" };
   267→
   268→      // ---------- Interactions ----------
   269→      case "show_interactions": {
   270→        const targetLoci = await resolveInteractionsTarget(viewer, cmd.target);
   271→        if (!targetLoci)
   272→          return { ok: false, detail: "Could not resolve interactions target" };
   273→        await showInteractionsAround(plugin, targetLoci, cmd.radius ?? 8);
   274→        return { ok: true, detail: `Interactions within ${cmd.radius ?? 8} Å` };
   275→      }
   276→
   277→      case "clear_interactions":
   278→        await clearInteractions(plugin);
   279→        return { ok: true, detail: "Interactions cleared" };
   280→
   281→      // ---------- Selection ----------
   282→      case "select": {
   283→        let loci: unknown = null;
   284→        if (cmd.target === "all") {
   285→          loci = getFirstStructureData(plugin);
   286→        } else if (cmd.target === "ligand") {
   287→          loci = await lociFromResidue(viewer, {});
   288→        } else {
   289→          loci = await lociFromResidue(viewer, cmd.target);
   290→        }
   291→        if (!loci) return { ok: false, detail: "Selection target not found" };
   292→        const action = cmd.action ?? "set";
   293→        // @ts-expect-error: action is union-typed at runtime
   294→        plugin.managers.structure.selection.modify(action, loci);
   295→        return { ok: true, detail: `Selected (${cmd.action ?? "set"})` };
   296→      }
   297→
   298→      case "clear_selection":
   299→        plugin.managers.structure.selection.clear();
   300→        plugin.managers.interactivity.lociSelects.deselectAll();
   301→        return { ok: true, detail: "Selection cleared" };
   302→
   303→      case "set_granularity":
   304→        plugin.managers.interactivity.setProps({ granularity: cmd.granularity });
   305→        return { ok: true, detail: `Granularity: ${cmd.granularity}` };
   306→
   307→      // ---------- Animation ----------
   308→      case "toggle_spin": {
   309→        setTrackballAnimate(plugin, "spin", { speed: cmd.speed ?? 0.1 });
   310→        return { ok: true, detail: "Spin animation enabled" };
   311→      }
   312→      case "toggle_rock": {
   313→        setTrackballAnimate(plugin, "rock", {});
   314→        return { ok: true, detail: "Rock animation enabled" };
   315→      }
   316→      case "stop_animation": {
   317→        setTrackballAnimate(plugin, undefined, {});
   318→        return { ok: true, detail: "Animation stopped" };
   319→      }
   320→
   321→      // ---------- Export ----------
   322→      case "export_snapshot": {
   323→        const dataUri = await plugin.helpers?.viewportScreenshot?.getImageDataUri(
   324→          {
   325→            width: cmd.width,
   326→            height: cmd.height,
   327→            transparency: false,
   328→            axes: true,
   329→          }
   330→        );
   331→        if (!dataUri)
   332→          return { ok: false, detail: "Screenshot failed" };
   333→        const a = document.createElement("a");
   334→        a.href = dataUri;
   335→        a.download = `molstar-snapshot-${Date.now()}.png`;
   336→        a.click();
   337→        return { ok: true, detail: "Snapshot exported", data: { dataUri } };
   338→      }
   339→
   340→      // ---------- Capture (returns data URI for embedding, no download) ----------
   341→      case "capture_snapshot": {
   342→        // Step 1: apply camera angle if requested
   343→        if (cmd.angle) {
   344→          try {
   345→            await applyCameraAngle(plugin, cmd.angle);
   346→          } catch (err) {
   347→            console.warn("[capture_snapshot] angle adjust failed:", err);
   348→          }
   349→        }
   350→        // Step 2: add text labels at residue positions
   351→        if (Array.isArray(cmd.labels) && cmd.labels.length > 0) {
   352→          for (const lbl of cmd.labels) {
   353→            try {
   354→              if (lbl.chain !== undefined && lbl.resno !== undefined) {
   355→                const loci = await lociFromResidue(viewer, {
   356→                  chain: lbl.chain,
   357→                  resno: lbl.resno,
   358→                });
   359→                if (loci) {
   360→                  await plugin.managers.structure.measurement.addLabel(loci, {
   361→                    customText: lbl.text ?? "",
   362→                  });
   363→                }
   364→              }
   365→            } catch (err) {
   366→              console.warn(
   367→                `[capture_snapshot] label "${lbl.text}" failed:`,
   368→                err
   369→              );
   370→            }
   371→          }
   372→          // Allow labels to render before capture
   373→          await new Promise((r) => setTimeout(r, 100));
   374→        }
   375→        // Step 3: capture
   376→        const dataUri =
   377→          await plugin.helpers?.viewportScreenshot?.getImageDataUri({
   378→            width: cmd.width ?? 1600,
   379→            height: cmd.height ?? 900,
   380→            transparency: false,
   381→            axes: true,
   382→          });
   383→        if (!dataUri)
   384→          return { ok: false, detail: "Capture failed" };
   385→        return {
   386→          ok: true,
   387→          detail: `Captured${cmd.label ? `: ${cmd.label}` : ""}${
   388→            cmd.angle ? ` (${cmd.angle})` : ""
   389→          }`,
   390→          data: {
   391→            dataUri,
   392→            label: cmd.label ?? "",
   393→            angle: cmd.angle ?? "",
   394→          },
   395→        };
   396→      }
   397→
   398→      // ---------- Alignment ----------
   399→      case "align_structures": {
   400→        const result = await alignStructures(
   401→          plugin,
   402→          cmd.ref,
   403→          cmd.mobile,
   404→          cmd.method ?? "tm-align"
   405→        );
   406→        return result;
   407→      }
   408→
   409→      // ---------- Misc ----------
   410→      case "set_background": {
   411→        const canvas3d = plugin.canvas3d as
   412→          | { setProps?: (fn: (p: unknown) => void) => void }
   413→          | undefined;
   414→        if (canvas3d?.setProps) {
   415→          canvas3d.setProps((p: unknown) => {
   416→            const props = p as { renderer?: { backgroundColor?: unknown } };
   417→            props.renderer = props.renderer ?? {};
   418→            props.renderer.backgroundColor = hexToNumber(cmd.color);
   419→          });
   420→        }
   421→        return { ok: true, detail: `Background: ${cmd.color}` };
   422→      }
   423→
   424→      case "toggle_component_visibility":
   425→        return { ok: true, detail: `Toggled ${cmd.component} (no-op stub)` };
   426→
   427→      // ---------- APBS Electrostatic 3D Visualization ----------
   428→      case "show_electrostatic_surface": {
   429→        const params: Record<string, unknown> = {};
   430→        if (cmd.chain) params.chain = cmd.chain;
   431→        if (cmd.ionicStrength) params.ionic_strength = cmd.ionicStrength;
   432→        const pdbId = cmd.pdbId ?? useAppStore.getState().structures[0]?.id ?? "";
   433→        const data = await runRecipe("apbs_electrostatic", pdbId || undefined, params);
   434→        const d = data?.data;
   435→        if (!d || d.error) {
   436→          return { ok: false, detail: `APBS failed: ${d?.error ?? "no data"}` };
   437→        }
   438→        // Apply molecular-surface + partial-charge color
   439→        const structs = plugin.managers.structure.hierarchy.current.structures;
   440→        if (structs.length > 0) {
   441→          try {
   442→            plugin.managers.structure.component.applyPreset([structs[0]], "molecular-surface");
   443→            await new Promise((r) => setTimeout(r, 200));
   444→            const comps = structs[0]?.components ?? [];
   445→            if (comps.length > 0) {
   446→              plugin.managers.structure.component.updateRepresentationsTheme(comps, {
   447→                color: "partial-charge",
   448→              });
   449→            }
   450→          } catch (e) {
   451→            console.warn("[show_electrostatic_surface] representation error:", e);
   452→          }
   453→        }
   454→        const viz: ElectrostaticViz = {
   455→          pdbId: pdbId || "unknown",
   456→          chainFilter: String(d.chain_filter ?? "all"),
   457→          ionicStrengthMm: Number(d.ionic_strength_mM ?? 150),
   458→          debyeLengthA: Number(d.debye_length_A ?? 0),
   459→          forcefield: String(d.forcefield ?? "PARSE"),
   460→          pdb2pqrUsed: Boolean(d.pdb2pqr_used),
   461→          numChargedAtoms: Number(d.num_charged_atoms ?? 0),
   462→          totalPotentialKcal: Number(d.total_potential_kcal_mol ?? 0),
   463→          meanPotentialKcal: Number(d.mean_potential_kcal_mol ?? 0),
   464→          mostStabilizing: (d.most_stabilizing ?? []).map((r: any) => ({
   465→            chain: String(r.chain ?? ""), resno: Number(r.resno ?? 0),
   466→            resname: String(r.resname ?? ""), atom: String(r.atom ?? ""),
   467→            charge: Number(r.charge ?? 0), potential_kcal_mol: Number(r.potential_kcal_mol ?? 0),
   468→          })),
   469→          mostDestabilizing: (d.most_destabilizing ?? []).map((r: any) => ({
   470→            chain: String(r.chain ?? ""), resno: Number(r.resno ?? 0),
   471→            resname: String(r.resname ?? ""), atom: String(r.atom ?? ""),
   472→            charge: Number(r.charge ?? 0), potential_kcal_mol: Number(r.potential_kcal_mol ?? 0),
   473→          })),
   474→          surfaceCharged: (d.surface_charged ?? d.all_results ?? []).slice(0, 30).map((r: any) => ({
   475→            chain: String(r.chain ?? ""), resno: Number(r.resno ?? 0),
   476→            resname: String(r.resname ?? ""), atom: String(r.atom ?? ""),
   477→            charge: Number(r.charge ?? 0), potential_kcal_mol: Number(r.potential_kcal_mol ?? 0),
   478→          })),
   479→          createdAt: Date.now(),
   480→        };
   481→        useAppStore.getState().setElectrostaticViz(viz);
   482→        return {
   483→          ok: true,
   484→          detail: `APBS surface: ${viz.numChargedAtoms} charged atoms, ${viz.totalPotentialKcal.toFixed(2)} kcal/mol (${viz.forcefield})`,
   485→          analysisResult: { kind: "recipe", recipe: "apbs_electrostatic", data },
   486→          data: viz,
   487→        };
   488→      }
   489→
   490→      // ---------- Druggability 3D Visualization ----------
   491→      case "show_druggable_pocket": {
   492→        const radius = cmd.radius ?? 8;
   493→        const pdbId = cmd.pdbId ?? useAppStore.getState().structures[0]?.id ?? "";
   494→        const data = await runRecipe("druggability", pdbId || undefined, {
   495→          ligandCompId: cmd.ligandCompId,
   496→          radius,
   497→        });
   498→        const d = data?.data;
   499→        if (!d || d.error) {
   500→          return { ok: false, detail: `Druggability failed: ${d?.error ?? "no data"}` };
   501→        }
   502→        // Focus on ligand + label pocket residues
   503→        try {
   504→          const ligandLoci = await lociFromResidue(viewer, { compId: cmd.ligandCompId });
   505→          if (ligandLoci) {
   506→            plugin.managers.camera.focusLoci(ligandLoci as any, { minRadius: 15 });
   507→          }
   508→        } catch (e) {
   509→          console.warn("[show_druggable_pocket] focus error:", e);
   510→        }
   511→        const residues = (d.residues ?? []).slice(0, 8);
   512→        for (const r of residues) {
   513→          try {
   514→            const loci = await lociFromResidue(viewer, { chain: r.chain, resno: r.resno });
   515→            if (loci) {
   516→              const catLabel = categoryLabel(r.category);
   517→              await plugin.managers.structure.measurement.addLabel(loci as any, {
   518→                customText: `${r.resname}${r.resno} (${catLabel})`,
   519→              });
   520→            }
   521→          } catch (e) {
   522→            console.warn("[show_druggable_pocket] label error:", e);
   523→          }
   524→        }
   525→        const viz: DruggabilityViz = {
   526→          pdbId: pdbId || "unknown",
   527→          ligand: cmd.ligandCompId,
   528→          radiusA: radius,

   635→      }
   636→    }
   637→  } catch (err) {
   638→    return {
   639→      ok: false,
   640→      detail: err instanceof Error ? err.message : String(err),
   641→    };
   642→  }

   671→/**
   672→ * Build a StructureElement.Loci from a residue spec by using the viewer's
   673→ * high-level `structureInteractivity` API to set the selection, then reading
   674→ * the resulting loci back from the selection manager.
   675→ *
   676→ * The prebuilt molstar bundle does NOT export `Script` or `MolScriptBuilder`,
   677→ * so we cannot build queries directly. The `viewer.structureInteractivity`
   678→ * helper accepts an `expression` callback that receives the MolScript builder
   679→ * `Q` internally — that's our portable entry point.
   680→ */
   681→async function lociFromResidue(
   682→  viewer: MolstarViewer,
   683→  ref: ResidueRef,
   684→  atomName?: string
   685→): Promise<unknown> {
   686→  const plugin = viewer.plugin;
   687→  const data = getFirstStructureData(plugin);
   688→  if (!data) return null;
   689→
   690→  // Build the MolScript expression. `Q` is provided by the viewer.
   691→  // atomGroups takes:
   692→  //   - 'residue-test': filters which residue groups to include
   693→  //   - 'atom-test': filters which atoms within each group to include
   694→  //   - 'group-by': how to group atoms (residueKey = per-residue)
   695→  const expr = (Q: any) => {
   696→    const residueTests: any[] = [];
   697→    if (ref.chain)
   698→      residueTests.push(
   699→        Q.core.rel.eq([
   700→          Q.struct.atomProperty.macromolecular.auth_asym_id
   701→            ? Q.struct.atomProperty.macromolecular.auth_asym_id()
   702→            : Q.struct.atomProperty.macromolecular.label_asym_id(),
   703→          ref.chain,
   704→        ])
   705→      );
   706→    if (ref.resno !== undefined)
   707→      residueTests.push(
   708→        Q.core.rel.eq([
   709→          Q.struct.atomProperty.macromolecular.auth_seq_id
   710→            ? Q.struct.atomProperty.macromolecular.auth_seq_id()
   711→            : Q.struct.atomProperty.macromolecular.label_seq_id(),
   712→          ref.resno,
   713→        ])
   714→      );
   715→    if (ref.compId)
   716→      residueTests.push(
   717→        Q.core.rel.eq([
   718→          Q.struct.atomProperty.macromolecular.label_comp_id(),
   719→          ref.compId,
   720→        ])
   721→      );
   722→    const residueTest =
   723→      residueTests.length === 0
   724→        ? undefined
   725→        : residueTests.length === 1
   726→        ? residueTests[0]
   727→        : Q.core.logic.and(residueTests);
   728→
   729→    // Atom name filter goes in 'atom-test', not 'residue-test'
   730→    const atomTest =
   731→      atomName || ref.atom
   732→        ? Q.core.rel.eq([
   733→            Q.struct.atomProperty.macromolecular.label_atom_id(),
   734→            atomName ?? ref.atom,
   735→          ])
   736→        : undefined;
   737→
   738→    const params: any = {
   739→      "group-by": Q.struct.atomProperty.macromolecular.residueKey(),
   740→    };
   741→    if (residueTest) params["residue-test"] = residueTest;
   742→    if (atomTest) params["atom-test"] = atomTest;
   743→    return Q.struct.generator.atomGroups(params);
   744→  };
   745→
   746→  try {
   747→    // Clear any existing selection, then use the viewer's high-level API to
   748→    // select everything matching the expression. The viewer handles building
   749→    // the loci internally; we then read it back from the selection manager.
   750→    plugin.managers.structure.selection.clear();
   751→    viewer.structureInteractivity({ expression: expr, action: ["select"] });
   752→    // The selection update may be synchronous or via a microtask. Wait a tick
   753→    // to let the state propagate before reading back.
   754→    await new Promise((r) => setTimeout(r, 30));
   755→    // Read back the loci from the selection manager.
   756→    // `entries` is a Map<Structure, { selection: Loci }> — iterate to find any
   757→    // non-empty entry.
   758→    const entries = plugin.managers.structure.selection.entries as
   759→      | Map<unknown, { selection?: { elements?: unknown[] } }>
   760→      | undefined;
   761→    if (entries && typeof entries.forEach === "function") {
   762→      let foundLoci: unknown = null;
   763→      entries.forEach((val) => {
   764→        if (
   765→          val?.selection?.elements &&
   766→          val.selection.elements.length > 0 &&
   767→          !foundLoci
   768→        ) {
   769→          foundLoci = val.selection;
   770→        }
   771→      });
   772→      if (foundLoci) return foundLoci;
   773→    }
   774→    // Fallback: try getLoci on the first structure
   775→    const loci = plugin.managers.structure.selection.getLoci(data);
   776→    if (loci && !isLociEmpty(loci)) {
   777→      return loci;
   778→    }
   779→    return null;
   780→  } catch (err) {
   781→    console.warn("[lociFromResidue] failed:", err);
   782→    return null;
   783→  }
   784→}
   785→
   786→/** Check if a Loci is empty (best-effort, structure-dependent). */
   787→function isLociEmpty(loci: unknown): boolean {
   788→  if (!loci) return true;
   789→  // Loci objects in molstar have an `elements` array; empty means no selection.
   790→  const l = loci as { elements?: unknown[] };
   791→  return !l.elements || l.elements.length === 0;
   792→}

   836→function setTrackballAnimate(
   837→  plugin: MolstarPlugin,
   838→  name: string | undefined,
   839→  params: { speed?: number }
   840→) {
   841→  const canvas3d = plugin.canvas3d as
   842→    | {
   843→        setProps?: (fn: (p: unknown) => void) => void;
   844→        props?: { trackball: { animate?: unknown } };
   845→      }
   846→    | undefined;
   847→  if (!canvas3d?.setProps) return;
   848→  canvas3d.setProps((p: unknown) => {
   849→    const props = p as {
   850→      trackball: {
   851→        animate?: { name: string; params: Record<string, unknown> };
   852→      };
   853→    };
   854→    props.trackball = props.trackball ?? {};
   855→    if (name) {
   856→      props.trackball.animate = { name, params };
   857→    } else {
   858→      props.trackball.animate = { name: "off", params: {} };
   859→    }
   860→  });

   863→/**
   864→ * Rotate the camera to one of four canonical angles before capturing.
   865→ *
   866→ * Strategy: reset to the default orientation, then apply a yaw rotation around
   867→ * the world Y axis (vertical) so that front/side/back face the camera. "top"
   868→ * rotates the camera to look down the Y axis by tweaking the trackball's
   869→ * `position`-like state via canvas3d.setProps.
   870→ *
   871→ * Since the prebuilt Molstar bundle doesn't expose `PluginContext.canvas3d.camera`
   872→ * in a stable way, we approximate by setting the trackball's animate spin to a
   873→ * one-shot rotation; for "top" we use a tilt via canvas3d props. This is a
   874→ * best-effort implementation — when more camera API is available, replace with
   875→ * direct camera matrix manipulation.
   876→ */
   877→async function applyCameraAngle(
   878→  plugin: MolstarPlugin,
   879→  angle: "front" | "side" | "top" | "back"
   880→): Promise<void> {
   881→  // Reset first to a known orientation
   882→  try {
   883→    plugin.managers.camera.reset();
   884→  } catch {
   885→    /* ignore */
   886→  }
   887→  // Allow the reset to settle
   888→  await new Promise((r) => setTimeout(r, 80));
   889→
   890→  const canvas3d = plugin.canvas3d as
   891→    | {
   892→        setProps?: (fn: (p: unknown) => void) => void;
   893→        camera?: { position?: unknown; up?: unknown; target?: unknown };
   894→      }
   895→    | undefined;
   896→  if (!canvas3d?.setProps) return;
   897→
   898→  // Apply a trackball spin to rotate around Y axis to the requested angle.
   899→  // "front" = no rotation (after reset)
   900→  // "side"  = 90° around Y
   901→  // "back"  = 180° around Y
   902→  // "top"   = look down the Y axis (rotate camera up over the structure)
   903→  if (angle === "front") {
   904→    // already front after reset
   905→    return;
   906→  }
   907→
   908→  // For non-front angles, use a brief spin animation that rotates the view.
   909→  // Since the trackball.animate spin rotates continuously, we use a one-shot
   910→  // approximation by setting the spin speed then clearing after a brief delay.
   911→  const spinDurationMs = 250;
   912→  setTrackballAnimate(plugin, "spin", { speed: angle === "top" ? 0.6 : 0.4 });
   913→
   914→  // After a brief moment, stop the spin. The viewer will have rotated ~90°.
   915→  await new Promise((r) => setTimeout(r, spinDurationMs));
   916→  setTrackballAnimate(plugin, undefined, {});
   917→
   918→  // For "top", also try to tilt the camera via canvas3d props (best-effort)
   919→  if (angle === "top") {
   920→    try {
   921→      canvas3d.setProps((p: unknown) => {
   922→        const props = p as { trackball?: { spin?: boolean; tilt?: number } };
   923→        // No stable API for tilt — leave as best-effort
   924→        void props;
   925→      });
   926→    } catch {
   927→      /* ignore */
   928→    }
   929→  }
   930→}
